/** Probador mínimo: acumula { nombre, ok, detalle }. */
function crearProbador_(resultados) {
  function anotar(nombre, ok, detalle) { resultados.push({ nombre: nombre, ok: !!ok, detalle: ok ? '' : String(detalle || '').slice(0, 600) }); }
  return {
    ok: function (nombre, condicion, detalle) { anotar(nombre, condicion, detalle || 'condición falsa'); },
    igual: function (nombre, real, esperado) {
      const ok = JSON.stringify(real) === JSON.stringify(esperado);
      anotar(nombre, ok, 'esperado ' + JSON.stringify(esperado) + ', obtenido ' + JSON.stringify(real));
    },
    lanza: function (nombre, fn, patron) {
      try { fn(); anotar(nombre, false, 'no lanzó error'); } catch (e) {
        const msg = String(e && e.message ? e.message : e);
        anotar(nombre, !patron || patron.test(msg), 'mensaje inesperado: ' + msg);
      }
    },
    seccion: function (nombre, fn) {
      try { fn(); } catch (e) { anotar(nombre + ' (error inesperado)', false, String(e && e.stack ? e.stack : e)); }
    },
  };
}

/**
 * Corre las pruebas. op = { corrida, email, soloMemoria }. Devuelve { total, ok, fallos, omitidas, detalle }.
 * Debe llamarse dentro de conBloqueo_ (las escrituras de los escenarios usan el mismo bloqueo).
 */
function testSistema_(ssVivo, tz, op) {
  op = op || {};
  const resultados = [];
  const t = crearProbador_(resultados);
  let omitidas = 0;

  // 1) En memoria (sin hoja)
  t.seccion('Núcleo', function () { pruebasNucleo_(t); });
  if (typeof pruebasCartera_ === 'function') t.seccion('Cartera', function () { pruebasCartera_(t); });
  if (typeof pruebasClientes_ === 'function') t.seccion('Clientes', function () { pruebasClientes_(t); });
  if (typeof pruebasProveedores_ === 'function') t.seccion('Proveedores', function () { pruebasProveedores_(t); });
  if (typeof pruebasComisiones_ === 'function') t.seccion('Comisiones', function () { pruebasComisiones_(t); });
  if (typeof pruebasTickets_ === 'function') t.seccion('Tickets', function () { pruebasTickets_(t); });
  if (typeof pruebasAgenda_ === 'function') t.seccion('Agenda', function () { pruebasAgenda_(t); });
  if (typeof pruebasFiscal_ === 'function') t.seccion('Fiscal y WhatsApp', function () { pruebasFiscal_(t); });

  // 2) Escenarios sobre una copia temporal del libro
  if (!op.soloMemoria) {
    if (tiempoRestante_() < 90000) {
      omitidas++;
      t.ok('Escenarios sobre copia temporal', true);
      resultados[resultados.length - 1].detalle = 'omitidos por tiempo';
    } else {
      escenariosEnCopia_(ssVivo, tz, t, function (contexto) {
        t.seccion('Escenarios · instalación', function () { escenariosInstalacion_(t, contexto); });
        t.seccion('Escenarios · seguridad', function () { escenariosSeguridad_(t, contexto); });
        if (typeof escenariosCartera_ === 'function') t.seccion('Escenarios · cartera', function () { escenariosCartera_(t, contexto); });
        if (typeof escenariosClientes_ === 'function') t.seccion('Escenarios · clientes', function () { escenariosClientes_(t, contexto); });
        if (typeof escenariosProveedores_ === 'function') t.seccion('Escenarios · proveedores', function () { escenariosProveedores_(t, contexto); });
        if (typeof escenariosComisiones_ === 'function') t.seccion('Escenarios · comisiones', function () { escenariosComisiones_(t, contexto); });
        if (typeof escenariosTickets_ === 'function') t.seccion('Escenarios · tickets', function () { escenariosTickets_(t, contexto); });
        if (typeof escenariosAgenda_ === 'function') t.seccion('Escenarios · agenda', function () { escenariosAgenda_(t, contexto); });
        if (typeof escenariosFiscal_ === 'function') t.seccion('Escenarios · fiscal', function () { escenariosFiscal_(t, contexto); });
      });
    }
  }

  // 3) Informe (lo único que se escribe en el libro vivo)
  const fallos = resultados.filter(function (r) { return !r.ok; });
  const corrida = op.corrida || 'Pruebas ' + Utilities.formatDate(new Date(), tz, 'yyyy-MM-dd HH:mm');
  try {
    anexarFilas_(ssVivo, 'instalacion', resultados.map(function (r) {
      return { fecha: new Date(), corrida: corrida, paso: 'Prueba · ' + r.nombre, recurso: 'testSistema', resultado: r.ok ? 'OK' : 'FALLO', detalle: r.detalle,
        accion: r.ok ? '' : 'Revisar (avisar a soporte con esta fila).', version: CONFIG.VERSION, usuario: op.email || '' };
    }), tz);
  } catch (e) { console.error('No se pudo escribir el informe de pruebas: ' + e); }
  const resumen = (resultados.length - fallos.length) + ' de ' + resultados.length + ' OK';
  try {
    PropertiesService.getScriptProperties().setProperty('MB_ULTIMAS_PRUEBAS', JSON.stringify({ fecha: Utilities.formatDate(new Date(), tz, 'yyyy-MM-dd HH:mm'), resumen: resumen, fallos: fallos.length }));
  } catch (e) { /* informativo */ }
  return { total: resultados.length, ok: resultados.length - fallos.length, fallos: fallos.length, omitidas: omitidas, detalle: resultados };
}

/**
 * Crea la copia temporal del libro, activa MODO_PRUEBA_ (servicios externos simulados) y garantiza la limpieza:
 * la copia va a la papelera, se revocan las sesiones de prueba y se borran las marcas de paquetes de la copia.
 */
function escenariosEnCopia_(ssVivo, tz, t, fn) {
  SpreadsheetApp.flush();
  const carpeta = subcarpeta_('Respaldos', true);
  const archivo = DriveApp.getFileById(ssVivo.getId()).makeCopy('PRUEBA TEMPORAL MB ' + Utilities.formatDate(new Date(), tz, 'yyyy-MM-dd HHmmss') + ' (se borra sola)', carpeta);
  const libro = SpreadsheetApp.openById(archivo.getId());
  const tokens = [];
  MODO_PRUEBA_ = { libro: libro, correos: [], whatsapp: [], calendar: [], sifen: [], archivos: [], compartidos: [], tokens: tokens };
  try {
    fn({ ss: libro, tz: tz, vivo: ssVivo, tokens: tokens });
  } finally {
    MODO_PRUEBA_ = null;
    reiniciarMemoriaEjecucion_();
    tokens.forEach(function (tk) { try { revocarSesion_(tk); } catch (e) { /* nada */ } });
    try {
      const props = PropertiesService.getScriptProperties();
      [CLAVE_PAQUETES, CLAVE_PAQUETES_AVISADOS].forEach(function (k) {
        const o = leerJsonPropiedad_(props, k);
        Object.keys(o).forEach(function (x) { if (x.indexOf(archivo.getId()) >= 0) delete o[x]; });
        props.setProperty(k, JSON.stringify(o));
      });
    } catch (e) { console.warn('Limpieza de marcas de prueba: ' + e); }
    try { CacheService.getScriptCache().remove('mb_usuarios_' + archivo.getId()); } catch (e) { /* nada */ }
    try { archivo.setTrashed(true); } catch (e) { console.error('No se pudo enviar a la papelera la copia de prueba: ' + e); }
  }
}

/** Llama a una función pública como lo haría la página (token de sesión, mes, idSolicitud único). */
function llamarPrueba_(fn, token, datos, extra) {
  const req = Object.assign({ token: token || '', mes: hoy_(CONFIG.ZONA_HORARIA).slice(0, 7), idSolicitud: 'prueba_' + Utilities.getUuid().replace(/-/g, '').slice(0, 20),
    datos: datos || {}, cliente: CONFIG.VERSION }, extra || {});
  return fn(req);
}

/** Sesión de prueba (se revoca al terminar). */
function sesionPrueba_(contexto, email) {
  const s = crearSesion_(email, false, 'PRUEBA');
  contexto.tokens.push(s.token);
  return s.token;
}

/** Política de referencia (copia del xlsx) para probar fórmulas sin depender de la hoja. */
function parametrosDeSemilla_() {
  const P = { _avisos: [], _faltan: [], _nombres: {}, _unidades: {} };
  semillaParametros_().forEach(function (p) {
    const r = interpretarParametro_(p[2], p[3]);
    P[p[0]] = r.ok ? r.valor : null;
    P._nombres[p[0]] = p[1];
    P._unidades[p[0]] = p[3];
  });
  return P;
}

function categoriasDeSemilla_() {
  return semillaCategorias_().map(function (c) {
    return { nombre: c[0], limite: c[1], entrega: c[2], cuotas: c[3], frecuencia: c[4],
      frecuencias: String(c[4]).toUpperCase().split(/\s*(?:,|;|\/|\bO\b)\s*/).map(function (s) { return s.trim(); }).filter(String), criterio: c[5] };
  });
}

  t.igual('Unidad de negocio: re-ejecutar no cambia nada', n2, 0);
  t.ok('Unidad de negocio: Ventas = DISTRIBUIDORA, Servicios realizados = SALÓN',
    leerTabla_(ss, 'ventas', tz).filas.every(function (v) { return v.unidad === 'DISTRIBUIDORA'; }) && leerTabla_(ss, 'realizados', tz).filas.every(function (r) { return r.unidad === 'SALÓN'; }));
  t.igual('Compatibilidad: ventas, servicios y productos existentes intactos',
    [leerTabla_(ss, 'ventas', tz).filas.length, leerTabla_(ss, 'realizados', tz).filas.length, leerTabla_(ss, 'productos', tz).filas.length], [antes.ventas, antes.realizados, antes.productos]);
  const filas = compartirYVerificar_(ss, tz).filter(function (f) { return f.resultado !== 'INFO'; });
  t.ok('Compartir con las administradoras: cada recurso verificado OK', filas.length > 0 && filas.every(function (f) { return f.resultado === 'OK'; }), JSON.stringify(filas));
  t.ok('Compartir no envía invitaciones por correo', MODO_PRUEBA_.correos.every(function (m) { return !/Invitaci/.test(m.subject || ''); }));
  const hoja = ss.getSheetByName(HOJA_CONFIG);
  const valores = hoja.getRange(1, 1, hoja.getLastRow(), 2).getValues();
  let fila = 0;
  valores.forEach(function (v, i) { if (String(v[0]).trim() === PARAM_REGISTRO) fila = i + 1; });
  if (fila) {
    const previo = hoja.getRange(fila, 2).getValue();
    hoja.getRange(fila, 2).setValue('');
    t.ok('Registro de nuevos correos sin valor = Cerrado (R6.3)', registroAbierto_(ss) === false);
    hoja.getRange(fila, 2).setValue(previo);
  }
}

function escenariosSeguridad_(t, c) {
  const ss = c.ss;
  const tz = c.tz;
  // Una operativa de prueba en «Usuarios» de la COPIA.
  const usuarios = leerTabla_(ss, 'usuarios', tz);
  if (!usuarios.filas.some(function (u) { return u.correo === CORREO_OPERATIVA_PRUEBA; })) {
    agregarFilas_(usuarios, [{ correo: CORREO_OPERATIVA_PRUEBA, activo: 'Sí', rol: ROLES.OPERATIVO, nombre: 'Operativa de prueba', notas: 'Solo en la copia de prueba' }], tz);
  }
  olvidarUsuariosEnCache_(ss);
  const tokAdmin = sesionPrueba_(c, CONFIG.CORREO_ADMIN);
  const tokOper = sesionPrueba_(c, CORREO_OPERATIVA_PRUEBA);

  const rOper = llamarPrueba_(obtenerDatos, tokOper);
  const json = JSON.stringify(rOper);
  t.ok('Operativa: el JSON no trae costos, márgenes, valor del inventario ni comisiones',
    !/"costo"|"costoUnitario"|"valorInventario"|"margen|"comision"|"porcentaje"|"proveedor"|"utilidad"/.test(json), json.slice(0, 300));
  t.ok('Operativa: no recibe el enlace a la hoja ni el panel de administración', !rOper.datos.admin && !/docs\.google\.com\/spreadsheets/.test(json));
  const rAdmin = llamarPrueba_(obtenerDatos, tokAdmin);
  t.ok('Administradora: recibe valor del inventario y comisiones', rAdmin.datos.kpis.valorInventario !== undefined && Array.isArray(rAdmin.datos.comisiones));
  let error = '';
  try { llamarPrueba_(guardarProducto, tokOper, { nombre: 'Prueba', precio: 1000 }); } catch (e) { error = e.message; }
  t.ok('Operativa: escritura reservada a administradoras rechazada en el servidor', /solo para administradoras/.test(error), error);
  error = '';
  try { llamarPrueba_(obtenerDatos, 'x'.repeat(64)); } catch (e) { error = e.message; }
  t.ok('Token de sesión falso rechazado con [SESION]', error.indexOf(PREFIJO_SESION) === 0, error);
  const vencida = crearSesion_(CONFIG.CORREO_ADMIN, false, 'PRUEBA');
  c.tokens.push(vencida.token);
  const clave = claveSesion_(vencida.token);
  const s = JSON.parse(PropertiesService.getScriptProperties().getProperty(clave));
  s.x = Date.now() - 1000;
  PropertiesService.getScriptProperties().setProperty(clave, JSON.stringify(s));
  CacheService.getScriptCache().remove(clave);
  error = '';
  try { llamarPrueba_(obtenerDatos, vencida.token); } catch (e) { error = e.message; }
  t.ok('Sesión vencida rechazada con [SESION]', error.indexOf(PREFIJO_SESION) === 0, error);
  error = '';
  try { iniciarSesion({ token: tokAdmin, mes: hoy_(tz).slice(0, 7), datos: {} }); } catch (e) { error = e.message; }
  t.ok('Página vieja (sin versión) recibe [SESION] y se le pide recargar', error.indexOf(PREFIJO_SESION) === 0 && /Recargue/.test(error), error);
  // Auditoría: una escritura deja rastro con correo verificado, función y unidad.
  const prod = leerTabla_(ss, 'productos', tz).filas.filter(function (p) { return esActivo_(p.activo); })[0];
  if (prod) {
    const hoja = ss.getSheetByName(ESQUEMA.auditoria.hoja);
    const antes = hoja.getLastRow();
    const id = 'prueba_' + Utilities.getUuid().replace(/-/g, '').slice(0, 16);
    const req = { token: tokAdmin, mes: hoy_(tz).slice(0, 7), idSolicitud: id, cliente: CONFIG.VERSION,
      datos: { tipo: 'ajuste_mas', codigo: prod.codigo, cantidad: 1, fecha: hoy_(tz), nota: 'Prueba automática (copia temporal)' } };
    const r1 = registrarMovimiento(req);
    const r2 = registrarMovimiento(req);
    t.ok('Doble envío (misma solicitud) no duplica el movimiento', r1.ok && r2.repetida === true);
    const aud = leerTabla_(ss, 'auditoria', tz).filas.slice(antes - 1);
    t.ok('Auditoría: alta con correo verificado, función, hoja y unidad', aud.some(function (a) {
      return a.usuario === CONFIG.CORREO_ADMIN && a.operacion === 'registrarMovimiento' && a.hoja === 'Movimientos de stock' && a.unidad === 'DISTRIBUIDORA' && a.solicitud === id;
    }), JSON.stringify(aud.slice(-3)));
  }
}
