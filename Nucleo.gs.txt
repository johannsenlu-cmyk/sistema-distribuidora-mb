/** Agrega las filas que falten en «Configuración» (sin tocar valores existentes) y actualiza la versión. */
function asegurarFilasConfiguracion_(ss) {
  const hoja = ss.getSheetByName(HOJA_CONFIG);
  if (!hoja) return;
  const ultima = Math.max(hoja.getLastRow(), 1);
  const valores = hoja.getRange(1, 1, ultima, 3).getValues();
  const filaDe = {};
  valores.forEach(function (f, i) { filaDe[String(f[0]).trim()] = i + 1; });
  if (filaDe[PARAM_REGISTRO]) hoja.getRange(filaDe[PARAM_REGISTRO], 3).setValue(DESCRIPCION_REGISTRO);
  const nuevas = [];
  filasConfiguracion_().forEach(function (f) {
    if (!filaDe[f[0]]) nuevas.push(f);
  });
  if (nuevas.length) {
    const inicio = ultima + 1;
    const fin = inicio + nuevas.length - 1;
    if (fin > hoja.getMaxRows()) hoja.insertRowsAfter(hoja.getMaxRows(), fin - hoja.getMaxRows() + 5);
    hoja.getRange(inicio, 1, nuevas.length, 3).setValues(nuevas.map(function (f) { return [f[0], celdaTexto_(f[1]), f[3]]; }));
    hoja.getRange(inicio, 3, nuevas.length, 1).setWrap(true);
    nuevas.forEach(function (f, i) {
      if (f[2]) hoja.getRange(inicio + i, 2).setDataValidation(reglaLista_(f[2]));
    });
  }
  memoria_.config = null;
  return nuevas.length;
}

function leerConfig_(ss, etiqueta) {
  if (!memoria_.config) {
    const mapa = {};
    const hoja = ss.getSheetByName(HOJA_CONFIG);
    if (hoja && hoja.getLastRow() >= 1) {
      hoja.getRange(1, 1, hoja.getLastRow(), 2).getValues().forEach(function (f) {
        const v = f[1] instanceof Date ? Utilities.formatDate(f[1], CONFIG.ZONA_HORARIA, 'yyyy-MM-dd') : f[1];
        mapa[String(f[0]).trim()] = typeof v === 'number' ? v : String(v).trim();
      });
    }
    memoria_.config = mapa;
  }
  return memoria_.config[etiqueta] === undefined ? '' : memoria_.config[etiqueta];
}

function escribirConfig_(ss, etiqueta, valor) {
  const hoja = ss.getSheetByName(HOJA_CONFIG);
  if (!hoja) throw errorUsuario_('Falta la pestaña «Configuración».');
  const v = hoja.getRange(1, 1, Math.max(hoja.getLastRow(), 1), 1).getValues();
  for (let i = 0; i < v.length; i++) {
    if (String(v[i][0]).trim() === etiqueta) {
      hoja.getRange(i + 1, 2).setValue(typeof valor === 'number' ? valor : celdaTexto_(valor));
      memoria_.config = null;
      return;
    }
  }
  throw errorUsuario_('Falta la fila «' + etiqueta + '» en «Configuración». Ejecute la instalación (instalarActualizacionMB) para crearla.');
}

/** «Sí» → true (acepta Si, SÍ, sí). */
function configSi_(ss, etiqueta) {
  return /^s[ií]/i.test(String(leerConfig_(ss, etiqueta) || '').trim());
}

/** Número de una fila de «Configuración» (acepta «30%» → 30). Vacío o inválido → null. */
function configNumero_(ss, etiqueta) {
  const v = leerConfig_(ss, etiqueta);
  if (v === '' || v === null || v === undefined) return null;
  if (typeof v === 'number') return v < 1 && /%/.test(String(etiqueta)) ? v * 100 : v;
  const n = decimal_(String(v).replace('%', ''));
  return isNaN(n) ? null : n;
}

/** 'SIMULACIÓN' (por defecto, ante cualquier duda) | 'PRODUCCIÓN' | 'APAGADO'. cual: 'sifen' | 'whatsapp'. */
function modoIntegracion_(ss, cual) {
  let v = '';
  try { v = String(leerConfig_(ss, cual === 'sifen' ? CFG.MODO_SIFEN : CFG.MODO_WA) || ''); } catch (e) { v = ''; }
  if (/^prod/i.test(v)) return 'PRODUCCIÓN';
  if (/^apag/i.test(v)) return 'APAGADO';
  return 'SIMULACIÓN';
}

/** Siembra las filas que falten (nunca modifica valores existentes). Devuelve { parametros, categorias }. */
function sembrarParametros_(ss, tz, email) {
  const tabla = leerTabla_(ss, 'parametros', tz);
  const ya = {};
  tabla.filas.forEach(function (f) { ya[clave_(f.clave)] = true; });
  const ahora = new Date();
  const nuevas = semillaParametros_().filter(function (p) { return !ya[p[0]]; }).map(function (p) {
    return { clave: p[0], nombre: p[1], valor: p[2], unidad: p[3], fuente: p[4], actualizado: ahora, usuario: email || USUARIO_ACTUALIZACION };
  });
  if (nuevas.length) {
    const desde = tabla.hoja.getLastRow() + 1;
    agregarFilas_(tabla, nuevas, tz);
    nuevas.forEach(function (p, i) {
      if (p.unidad === '%' && typeof p.valor === 'number') tabla.hoja.getRange(desde + i, tabla.idx.valor + 1).setNumberFormat('0.00%');
      if (/^Gs/.test(p.unidad) && typeof p.valor === 'number') tabla.hoja.getRange(desde + i, tabla.idx.valor + 1).setNumberFormat('#,##0');
    });
  }
  const cats = leerTabla_(ss, 'categorias', tz);
  let nCat = 0;
  if (!cats.filas.length) {
    const desde = cats.hoja.getLastRow() + 1;
    agregarFilas_(cats, semillaCategorias_().map(function (c) {
      return { nombre: c[0], limite: c[1], entrega: c[2], cuotas: c[3], frecuencia: c[4], criterio: c[5], activo: 'Sí' };
    }), tz);
    cats.hoja.getRange(desde, cats.idx.entrega + 1, 4, 1).setNumberFormat('0%');
    nCat = 4;
  }
  memoria_.parametros = null;
  memoria_.categorias = null;
  return { parametros: nuevas.length, categorias: nCat };
}

/** Interpreta un valor de la hoja según su unidad. Devuelve { valor, ok }. */
function interpretarParametro_(v, unidad) {
  if (/^(texto|código|AAAA-MM)$/i.test(unidad)) {
    const s = v instanceof Date ? Utilities.formatDate(v, CONFIG.ZONA_HORARIA, 'yyyy-MM') : String(v === null || v === undefined ? '' : v).trim();
    return { valor: s, ok: s !== '' };
  }
  if (/^s[ií]\/no$/i.test(unidad)) {
    const s = String(v || '').trim();
    return { valor: /^s[ií]/i.test(s), ok: /^(s[ií]|no)$/i.test(s) };
  }
  if (v === '' || v === null || v === undefined) return { valor: null, ok: false };
  let n;
  if (typeof v === 'number') n = v;
  else if (unidad === '%') n = decimal_(String(v).replace(/[^0-9,.\-%]/g, '').replace('%', ''));
  else n = entero_(v); // Gs., días, cuotas: «15.000.000» → 15000000
  if (isNaN(n) || n < 0) return { valor: null, ok: false };
  if (unidad === '%') {
    // En la hoja el % se guarda como fracción (0,3). Si alguien escribió «30» se interpreta como 30%.
    if (typeof v !== 'number' && /%/.test(String(v))) n = n / 100;
    else if (n >= 1) n = n / 100;
  }
  return { valor: n, ok: true };
}

/**
 * Política vigente: { P_xxx: valor, _avisos: [...], _faltan: [...], _nombres: {clave: parámetro} }.
 * % como fracción (0,3). Lo vacío o inválido NO se reemplaza: queda en _faltan y param_() lo exige.
 */
function leerParametros_(ss, tz) {
  if (memoria_.parametros) return memoria_.parametros;
  let filas = [];
  try { filas = leerTabla_(ss, 'parametros', tz).filas; } catch (e) { filas = []; }
  const P = { _avisos: [], _faltan: [], _nombres: {}, _unidades: {} };
  const semilla = {};
  semillaParametros_().forEach(function (p) { semilla[p[0]] = p; });
  const vistos = {};
  filas.forEach(function (f) {
    const k = clave_(f.clave);
    if (!k) return;
    vistos[k] = true;
    const unidad = String(f.unidad || (semilla[k] ? semilla[k][3] : 'texto')).trim();
    const r = interpretarParametro_(f.valor, unidad);
    P._nombres[k] = f.nombre || k;
    P._unidades[k] = unidad;
    if (r.ok) P[k] = r.valor;
    else {
      P[k] = null;
      if (PARAMETROS_OPCIONALES_.indexOf(k) < 0) {
        P._faltan.push(k);
        P._avisos.push('Parámetro «' + (f.nombre || k) + '» vacío o inválido en «Parámetros de crédito».');
      }
    }
  });
  Object.keys(semilla).forEach(function (k) {
    if (vistos[k]) return;
    P[k] = null;
    P._nombres[k] = semilla[k][1];
    P._unidades[k] = semilla[k][3];
    if (PARAMETROS_OPCIONALES_.indexOf(k) < 0) P._faltan.push(k);
  });
  if (P.P_TEA_FIN !== null && P.P_LIMITE_BCP !== null && P.P_TEA_MORA !== null &&
      Math.max(P.P_TEA_FIN, P.P_TEA_MORA) > P.P_LIMITE_BCP + 1e-12) {
    P._avisos.push('REVISAR: una tasa supera el límite BCP');
  }
  if (P.P_IVA_RECARGO === null) P._avisos.push('IVA del recargo financiero y de la mora: PENDIENTE (se usa la tasa general).');
  memoria_.parametros = P;
  return P;
}

/** Valor obligatorio de la política. Si falta, detiene la operación con un mensaje claro (nunca usa un valor del código). */
function param_(P, clave) {
  const v = P[clave];
  if (v === null || v === undefined || (typeof v === 'number' && isNaN(v))) {
    throw errorUsuario_('Falta el parámetro «' + (P._nombres[clave] || clave) + '» en «Parámetros de crédito». Complételo (Sistema › Parámetros) y vuelva a intentar.');
  }
  return v;
}

/** Tasa general de IVA en puntos (10). */
function ivaGeneral_(P) {
  return Math.round(param_(P, 'P_IVA') * 1000) / 10;
}

function leerCategorias_(ss, tz) {
  if (memoria_.categorias) return memoria_.categorias;
  let filas = [];
  try { filas = leerTabla_(ss, 'categorias', tz).filas; } catch (e) { filas = []; }
  const lista = filas.filter(function (c) { return clave_(c.nombre) && esActivo_(c.activo); }).map(function (c) {
    const entrega = Number(c.entrega) || 0;
    return {
      nombre: clave_(c.nombre), limite: Number(c.limite) || 0, entrega: entrega >= 1 ? entrega / 100 : entrega, cuotas: Number(c.cuotas) || 0,
      frecuencia: String(c.frecuencia || ''),
      frecuencias: String(c.frecuencia || '').toUpperCase().split(/\s*(?:,|;|\/|\bO\b)\s*/).map(function (s) { return s.trim(); }).filter(String),
      criterio: c.criterio,
    };
  });
  memoria_.categorias = lista;
  return lista;
}

// ─────────────────────── Redondeos (idénticos al xlsx) ───────────────────────

/** Normaliza a 15 cifras significativas, como Excel, antes de redondear (evita 207.000,00000000003). */
function excel15_(x) {
  return Number(Number(x).toPrecision(15));
}

/** REDONDEAR.MAS / ROUNDUP al múltiplo indicado (↑1.000). */
function redondearArriba_(x, multiplo) {
  const m = multiplo || 1;
  const q = excel15_(x / m);
  return (q >= 0 ? Math.ceil(q) : -Math.ceil(-q)) * m;
}

/** REDONDEAR.MENOS / ROUNDDOWN al múltiplo indicado (↓1.000). */
function redondearAbajo_(x, multiplo) {
  const m = multiplo || 1;
  const q = excel15_(x / m);
  return (q >= 0 ? Math.floor(q) : -Math.floor(-q)) * m;
}

/** REDONDEAR / ROUND al múltiplo indicado (mitad hacia afuera, como Excel). */
function redondear_(x, multiplo) {
  const m = multiplo || 1;
  const q = excel15_(x / m);
  return (q >= 0 ? Math.floor(q + 0.5) : -Math.floor(-q + 0.5)) * m;
}

/** Guaraníes enteros, mitad hacia arriba (comisiones, como la liquidación vigente: Math.round). */
function gsEnteros_(x) {
  return Math.round(excel15_(x));
}

// ─────────────────────── IVA (10%, 5%, exenta; precios con IVA incluido) ───────────────────────

/** Tasa en puntos (10, 5 o 0) desde la columna «Tasa IVA»; vacío → tasa general de la política. Acepta '10%', 0.1, 'Exenta'. */
function tasaIva_(valor, P) {
  const s = String(valor === null || valor === undefined ? '' : valor).trim().toLowerCase();
  if (!s) return P ? ivaGeneral_(P) : 10;
  if (/^ex/.test(s)) return 0;
  const n = decimal_(s.replace('%', ''));
  if (n === 0) return 0;
  if (n === 5 || n === 0.05) return 5;
  if (n === 10 || n === 0.1) return 10;
  throw errorUsuario_('Tasa de IVA no válida («' + valor + '»): use 10, 5 o 0 (exenta).');
}

/** Texto para la columna «Tasa IVA». */
function textoTasa_(n) {
  return String(n === 0 ? 0 : (n === 5 ? 5 : 10));
}

/** IVA incluido en un importe: REDONDEAR(total × tasa ÷ (1 + tasa); 0) → 10% = ÷ 11, 5% = ÷ 21, exenta = 0. */
function ivaIncluido_(total, tasa) {
  if (!tasa) return 0;
  return redondear_(Number(total) * tasa / (100 + tasa), 1);
}

/** Desglose fiscal de líneas { total (con IVA), tasa }: gravadas (con IVA), IVA por tasa, exenta y neto. */
function desgloseIva_(lineas) {
  const r = { total: 0, gravada10: 0, iva10: 0, gravada5: 0, iva5: 0, exenta: 0 };
  lineas.forEach(function (l) {
    const t = Number(l.total) || 0;
    r.total += t;
    if (l.tasa === 10) r.gravada10 += t;
    else if (l.tasa === 5) r.gravada5 += t;
    else r.exenta += t;
  });
  r.iva10 = ivaIncluido_(r.gravada10, 10);
  r.iva5 = ivaIncluido_(r.gravada5, 5);
  r.neto = r.total - r.iva10 - r.iva5;
  return r;
}

// ─────────────────────── Precios: descuento, margen, mínimos (R0.2, R0.3, R4.7) ───────────────────────

/** Descuento sobre el precio de lista (fracción). Sin lista o precio ≥ lista → 0. */
function descuentoLinea_(precio, lista) {
  if (!(lista > 0) || precio >= lista) return 0;
  return 1 - precio / lista;
}

/** Margen = (neto − costo) ÷ neto, con neto = total − IVA incluido. Sin costo → null. */
function margenLinea_(total, tasa, costoTotal, P) {
  if (!(costoTotal > 0) || !(total > 0)) return null;
  const neto = total - ivaIncluido_(total, tasa);
  const costoSinIva = P && P.P_COSTO_CON_IVA === true ? costoTotal / (1 + tasa / 100) : costoTotal;
  return (neto - costoSinIva) / neto;
}

/** Precio mínimo con IVA = ↑1.000 [costo ÷ (1 − P_MARGEN_MIN) × (1 + tasa)]. */
function precioMinimo_(costo, tasa, P) {
  if (!(costo > 0)) return null;
  return redondearArriba_(costo / (1 - param_(P, 'P_MARGEN_MIN')) * (1 + tasa / 100), param_(P, 'P_RED_PRECIO'));
}

/** Precio sugerido = redondeo a 1.000 [costo ÷ (1 − P_MARGEN_OBJ) × (1 + tasa)]. */
function precioSugerido_(costo, tasa, P) {
  if (!(costo > 0)) return null;
  return redondear_(costo / (1 - param_(P, 'P_MARGEN_OBJ')) * (1 + tasa / 100), param_(P, 'P_RED_PRECIO'));
}

function esEquipamiento_(item, P) {
  const linea = normalizar_(item.linea || '');
  if (linea && P && P.P_LINEA_EQUIPO && linea === normalizar_(P.P_LINEA_EQUIPO)) return true;
  return /^equipo|equipamiento|mobiliario/.test(normalizar_(item.categoria || ''));
}

/**
 * Controles de precio de TODA venta (mensajes literales del xlsx).
 * lineas: [{ nombre, tipo: 'PRODUCTO'|'SERVICIO'|'SISTEMA', precio, lista, total, tasa, costoTotal }].
 * Devuelve [{ codigo, mensaje, autorizable, linea }]: lo autorizable lo aprueba una administradora con motivo;
 * lo no autorizable se rechaza siempre. Los mensajes no revelan costos (los ve también una operativa).
 */
function controlesPrecios_(lineas, P) {
  const problemas = [];
  const vend = param_(P, 'P_DESC_VEND');
  const max = param_(P, 'P_DESC_MAX');
  const margenMin = param_(P, 'P_MARGEN_MIN');
  lineas.forEach(function (l, i) {
    if (l.tipo === 'SISTEMA') return;
    const desc = descuentoLinea_(l.precio, l.lista);
    if (desc > max + 1e-9) {
      problemas.push({ codigo: 'DESCUENTO FUERA DE POLÍTICA', autorizable: false, linea: i,
        mensaje: 'DESCUENTO FUERA DE POLÍTICA: «' + l.nombre + '» tiene ' + porcentajeTexto_(desc) + ' (máximo ' + porcentajeTexto_(max) + ').' });
    } else if (desc > vend + 1e-9) {
      problemas.push({ codigo: 'DESCUENTO REQUIERE AUTORIZACIÓN', autorizable: true, linea: i,
        mensaje: 'DESCUENTO REQUIERE AUTORIZACIÓN: «' + l.nombre + '» tiene ' + porcentajeTexto_(desc) + ' (la vendedora puede hasta ' + porcentajeTexto_(vend) + ').' });
    }
    if (l.tipo === 'PRODUCTO') {
      const m = margenLinea_(l.total, l.tasa, l.costoTotal, P);
      if (m !== null && m < margenMin - 1e-9) {
        problemas.push({ codigo: 'MARGEN BAJO MÍNIMO', autorizable: true, linea: i, mensaje: 'MARGEN BAJO MÍNIMO: «' + l.nombre + '» (requiere administradora).' });
      }
    }
  });
  return problemas;
}

function porcentajeTexto_(fraccion) {
  return numeroTexto_(Math.round(Number(fraccion) * 1000) / 10) + '%';
}

function normalizarMedioPago_(m) {
  const k = normalizar_(m);
  const encontrado = LISTAS.MEDIOS_PAGO.filter(function (x) { return normalizar_(x) === k; })[0];
  if (!encontrado) throw errorUsuario_('FALTA MEDIO DE PAGO (opciones: ' + LISTAS.MEDIOS_PAGO.join(', ') + ').');
  return encontrado;
}

/**
 * Valida pagos de contado: [{ medio, monto, operacion, recibo }] → normalizados. Exige medio, N° de operación si no es
 * efectivo y que la suma sea igual al total. Mensajes literales del xlsx.
 */
function validarPagos_(pagos, total) {
  const lista = Array.isArray(pagos) ? pagos : [];
  if (!lista.length) throw errorUsuario_('FALTA MEDIO DE PAGO');
  let suma = 0;
  const salida = lista.map(function (p, i) {
    if (!p || !String(p.medio || '').trim()) throw errorUsuario_('FALTA MEDIO DE PAGO');
    const medio = normalizarMedioPago_(p.medio);
    const monto = requerirEntero_(p.monto, 'Monto del pago ' + (i + 1), 1, LIMITES.PRECIO_UNITARIO * 10);
    const operacion = texto_(p.operacion, 40);
    if (medio !== 'EFECTIVO' && !operacion) throw errorUsuario_('FALTA N° DE OPERACIÓN DEL PAGO (' + medio + ').');
    suma += monto;
    return { medio: medio, monto: monto, operacion: operacion, recibo: texto_(p.recibo, 30) };
  });
  if (suma !== total) {
    throw errorUsuario_('Los pagos suman Gs. ' + miles_(suma) + ' y el total es Gs. ' + miles_(total) + ': deben coincidir.');
  }
  return salida;
}

// ─────────────────────── Identificación: RUC, CI, celular, correo ───────────────────────

/**
 * Dígito verificador del RUC (módulo 11 de la SET, base máxima 11): de derecha a izquierda se multiplica por 2, 3, … 11
 * y se vuelve a 2; resto = suma mod 11; DV = resto > 1 ? 11 − resto : 0. Verificado con 1307114-9 y 4581379-5 (libro vivo).
 */
function dvRuc_(base) {
  const s = String(base).toUpperCase().split('').map(function (ch) { return /\d/.test(ch) ? ch : String(ch.charCodeAt(0)); }).join('');
  let k = 2;
  let total = 0;
  for (let i = s.length - 1; i >= 0; i--) {
    total += Number(s.charAt(i)) * k;
    k = k >= 11 ? 2 : k + 1;
  }
  const resto = total % 11;
  return resto > 1 ? 11 - resto : 0;
}

/**
 * RUC con formato 1234567-8 (sin puntos). Con «Validar dígito verificador del RUC» = Sí exige el DV correcto.
 * Devuelve el RUC normalizado ('' si viene vacío). opciones.validarDv: boolean.
 */
function validarRuc_(valor, opciones) {
  const s = texto_(valor, 20).toUpperCase().replace(/\s+/g, '').replace(/\./g, '');
  if (!s) return '';
  const m = s.match(/^(\d{3,9})-(\d)$/);
  if (!m) throw errorUsuario_('El RUC «' + s + '» no es válido. Use el formato 1234567-8 (sin puntos), con el dígito verificador.');
  const validar = !opciones || opciones.validarDv !== false;
  if (validar && dvRuc_(m[1]) !== Number(m[2])) {
    throw errorUsuario_('El dígito verificador del RUC «' + s + '» no coincide (debería ser ' + m[1] + '-' + dvRuc_(m[1]) + '). Revíselo en la constancia de RUC.');
  }
  return m[1] + '-' + m[2];
}

/** RUC del libro (sin lanzar): { ruc, error }. */
function revisarRuc_(valor, validarDv) {
  try { return { ruc: validarRuc_(valor, { validarDv: validarDv }), error: '' }; } catch (e) { return { ruc: '', error: e.message }; }
}

/** Celular paraguayo normalizado a 595 + 9 dígitos (5959XXXXXXXX). Acepta 0981…, 981…, +595 981…. Inválido → ''. */
function normalizarCelular_(valor) {
  let d = String(valor === null || valor === undefined ? '' : valor).replace(/\D/g, '');
  if (!d) return '';
  if (d.indexOf('00595') === 0) d = d.slice(2);
  if (d.indexOf('595') === 0) d = d.slice(3);
  if (d.charAt(0) === '0') d = d.slice(1);
  return /^9\d{8}$/.test(d) ? '595' + d : '';
}

/** Compatibilidad con la v2.2 (Whatsapp.gs): mismo criterio. */
function normalizarWhatsapp_(valor) {
  return normalizarCelular_(valor);
}

function normalizarCi_(valor) {
  const d = String(valor === null || valor === undefined ? '' : valor).replace(/[.\s]/g, '').trim();
  if (!d) return '';
  if (!/^\d{3,10}$/.test(d)) throw errorUsuario_('La CI «' + valor + '» no es válida: escriba solo números, sin puntos.');
  return d;
}

function correoValido_(valor) {
  return /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(String(valor || '').trim());
}

function modoFacturacion_(ss) {

}

/** Entidad fiscal que emite para una unidad: UNIFICADO → siempre DISTRIBUIDORA; SEPARADO → la de la unidad. */
function entidadFiscalDe_(ss, tz, unidad) {
  const modo = modoFacturacion_(ss);
  const objetivo = modo === 'SEPARADO' ? unidadValida_(unidad, 'DISTRIBUIDORA') : 'DISTRIBUIDORA';
  let filas = [];
  try { filas = leerTabla_(ss, 'entidades', tz).filas; } catch (e) { filas = []; }
  const fila = filas.filter(function (f) { return unidadValida_(f.unidad, '') === objetivo; })[0] || null;
  const faltan = [];
  if (!fila) faltan.push('fila de ' + objetivo);
  else {
    if (!fila.razonSocial) faltan.push('Razón social');
    if (!fila.ruc) faltan.push('RUC');
    if (!fila.timbrado) faltan.push('Timbrado');
    if (!fila.establecimiento) faltan.push('Establecimiento');
    if (!fila.punto) faltan.push('Punto de expedición');
  }
  return { modo: modo, unidad: objetivo, fila: fila, completa: !faltan.length, faltan: faltan };
}

/** Propone la línea desde la «Categoría». Lo no mapeable (Blanqueria, CORPORAL, Saldos, Ocasional…) → REVISAR. */
function lineaPropuesta_(categoria, esServicio) {
  if (esServicio) return 'SERVICIOS Y CAPACITACIÓN';
  const c = normalizar_(categoria);
  if (/^herramienta/.test(c)) return 'HERRAMIENTAS Y ACCESORIOS';

  if (c === 'facial') return 'FACIAL Y ROSTRO';
  if (/^equipo/.test(c)) return 'EQUIPAMIENTO Y MOBILIARIO';
  return 'REVISAR';
}

/**
 * Envía un correo con MailApp. En modo prueba no envía (lo anota). Si la cuota diaria no alcanza, no envía y lo informa.
 * o: { to, subject, body, htmlBody, attachments }. Devuelve { ok, simulado, motivo }.
 */
function enviarCorreo_(o) {
  const destinatarios = String(o.to || '').split(',').map(function (s) { return s.trim(); }).filter(String);
  if (!destinatarios.length) return { ok: false, motivo: 'sin destinatario' };
  if (MODO_PRUEBA_) {
    MODO_PRUEBA_.correos.push({ to: destinatarios.join(','), subject: o.subject });
    return { ok: true, simulado: true };
  }
  let cuota = 0;
  try { cuota = MailApp.getRemainingDailyQuota(); } catch (e) { cuota = 0; }
  if (cuota < destinatarios.length) {
    console.warn('Cuota de MailApp insuficiente (' + cuota + '): no se envió «' + o.subject + '».');
    return { ok: false, motivo: 'cuota de correo agotada por hoy (' + cuota + ')' };
  }
  MailApp.sendEmail(Object.assign({ name: 'Distribuidora MB' }, o, { to: destinatarios.join(',') }));
  return { ok: true };
}

// ─────────────────────── Carpeta del sistema en Drive (R6.1 paso 3) ───────────────────────

/** Carpeta «Distribuidora MB» de la cuenta propietaria (por ID guardado; si falta y crear = true, la crea en Mi unidad). */
function carpetaSistema_(crear) {
  const props = PropertiesService.getScriptProperties();
  const id = props.getProperty('MB_ID_CARPETA');
  if (id) {
    try {
      const c = DriveApp.getFolderById(id);
      if (!c.isTrashed()) return c;
    } catch (e) { /* se busca o se crea */ }
  }
  const raiz = DriveApp.getRootFolder();
  const it = raiz.getFoldersByName(CONFIG.CARPETA_SISTEMA);
  let carpeta = it.hasNext() ? it.next() : null;
  if (!carpeta) {
    if (!crear) return null;
    carpeta = raiz.createFolder(CONFIG.CARPETA_SISTEMA);
  }
  props.setProperty('MB_ID_CARPETA', carpeta.getId());
  return carpeta;
}

/** Subcarpeta (Respaldos, Proveedores, Importaciones, Comprobantes, Exportaciones). */
function subcarpeta_(nombre, crear) {
  const base = carpetaSistema_(crear !== false);
  if (!base) return null;
  const it = base.getFoldersByName(nombre);
  if (it.hasNext()) return it.next();
  return crear === false ? null : base.createFolder(nombre);
}

/** Guarda un archivo generado por el sistema en su subcarpeta. Devuelve { id, url, nombre }. */
function guardarArchivoSistema_(blob, nombreSubcarpeta, nombre) {
  if (MODO_PRUEBA_) {
    MODO_PRUEBA_.archivos.push({ carpeta: nombreSubcarpeta, nombre: nombre });
    return { id: 'prueba', url: '', nombre: nombre };
  }
  const carpeta = subcarpeta_(nombreSubcarpeta, true);
  const archivo = carpeta.createFile(blob.setName(nombre));
  return { id: archivo.getId(), url: archivo.getUrl(), nombre: nombre };
}

// ─────────────────────── Respaldos (antes de toda migración y diario) ───────────────────────

/** Copia completa del libro en «Distribuidora MB/Respaldos» (las copias no heredan el compartido). */
function respaldoAhora_(ss, tz, etiqueta) {
  SpreadsheetApp.flush();
  const carpeta = subcarpeta_('Respaldos', true);
  const nombre = 'Respaldo MB ' + Utilities.formatDate(new Date(), tz, 'yyyy-MM-dd HHmm') + (etiqueta ? ' (' + etiqueta + ')' : '');
  const copia = DriveApp.getFileById(ss.getId()).makeCopy(nombre, carpeta);
  const info = { nombre: nombre, id: copia.getId(), fecha: Utilities.formatDate(new Date(), tz, 'yyyy-MM-dd HH:mm') };
  PropertiesService.getScriptProperties().setProperty('MB_ULTIMO_RESPALDO', JSON.stringify(info));
  return { nombre: nombre, id: copia.getId(), carpeta: carpeta.getName(), borrados: limpiarRespaldos_(carpeta) };
}

let RESPALDO_EJECUCION_ = null;

/** Respaldo «antes de …» una sola vez por ejecución (lo usan los paquetes y las migraciones). */
function respaldoUnaVez_(ss, tz, motivo) {
  if (MODO_PRUEBA_) return { nombre: '(prueba: sin respaldo)', id: '' };
  if (!RESPALDO_EJECUCION_) RESPALDO_EJECUCION_ = respaldoAhora_(ss, tz, 'antes de ' + motivo);
  return RESPALDO_EJECUCION_;
}

/** Envía a la papelera los respaldos DIARIOS con más de 14 días. Los respaldos «(antes de …)» se conservan. */
function limpiarRespaldos_(carpeta) {
  const limite = Date.now() - CONFIG.RETENCION_RESPALDOS_DIAS * 86400000;
  let n = 0;
  const it = carpeta.getFiles();
  while (it.hasNext()) {
    const f = it.next();
    const nombre = f.getName();
    if (nombre.indexOf('Respaldo MB ') !== 0 || /\(antes de/i.test(nombre)) continue;
    if (f.getDateCreated().getTime() < limite) { f.setTrashed(true); n++; }
  }
  return n;
}

// ─────────────────────── Compartir sin invitación (servicio avanzado de Drive v3) ───────────────────────

/**
 * Comparte un archivo o carpeta con EDICIÓN sin enviar correo de invitación.
 * Drive API v3: Permissions.create(recurso, fileId, { sendNotificationEmail: false }) — el servicio avanzado «Drive»
 * debe estar activado (appsscript.json). DriveApp.addEditor no se usa porque puede notificar.
 * Devuelve 'OK' | 'YA TENÍA' | 'ERROR: …'.
 */
function compartirSinAviso_(idArchivo, correo, rol) {
  if (MODO_PRUEBA_) { MODO_PRUEBA_.compartidos.push({ id: idArchivo, correo: correo }); return 'OK'; }
  if (typeof Drive === 'undefined' || !Drive.Permissions) {
    return 'ERROR: el servicio avanzado «Drive» no está activado (Editor › Servicios › Drive API v3).';
  }
  const actuales = permisosDe_(idArchivo);
  if (actuales.error) return 'ERROR: ' + actuales.error;
  const ya = actuales.lista.filter(function (p) { return p.correo === correo; })[0];
  if (ya && (ya.rol === 'writer' || ya.rol === 'owner' || ya.rol === 'organizer' || ya.rol === 'fileOrganizer')) return 'YA TENÍA';
  try {
    Drive.Permissions.create({ type: 'user', role: rol || 'writer', emailAddress: correo }, idArchivo, { sendNotificationEmail: false });
    return 'OK';
  } catch (e) {
    return 'ERROR: ' + (e && e.message ? e.message : e);
  }
}

/** Permisos reales de un archivo (relectura para verificar). { lista: [{ correo, rol }], error }. */
function permisosDe_(idArchivo) {
  if (MODO_PRUEBA_) return { lista: MODO_PRUEBA_.compartidos.filter(function (c) { return c.id === idArchivo; }).map(function (c) { return { correo: c.correo, rol: 'writer' }; }), error: '' };
  try {
    const r = Drive.Permissions.list(idArchivo, { fields: 'permissions(emailAddress,role,type)' });
    const lista = (r && r.permissions ? r.permissions : []).filter(function (p) { return p.type === 'user'; }).map(function (p) {
      return { correo: String(p.emailAddress || '').toLowerCase(), rol: p.role };
    });
    return { lista: lista, error: '' };
  } catch (e) {
    return { lista: [], error: String(e && e.message ? e.message : e) };
  }
}

function enmascarar_(valor, secreto) {
  const s = String(valor || '');
  if (!s) return '';
  if (!secreto) return s;
  return s.length <= 6 ? '••••' : '••••••' + s.slice(-4);
}
