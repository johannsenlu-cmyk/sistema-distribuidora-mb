/**
 * v2.3: la página se arma como plantilla (createTemplateFromFile) para incluir los parciales UI_* e inyectar
 * el «arranque» del ingreso: enlace de MB Acceso, sesión recién verificada por Google (?acceso=…) o el error.
 * ?page=health abre el panel «Sistema › Salud» (solo administradoras, tras ingresar).
 */
function doGet(e) {
  const p = e && e.parameter ? e.parameter : {};
  const t = HtmlService.createTemplateFromFile('Index');
  const pagina = String(p.page || '');
  t.pagina = /^[a-z]{1,20}$/.test(pagina) ? pagina : '';
  t.version = CONFIG.VERSION;
  t.arranque = jsonParaHtml_(arranqueIngreso_(p));
  return t.evaluate()
    .setTitle('Distribuidora MB · Gestión')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1');
}

/**
 * Datos de arranque de la página (nunca lanza). Con ?acceso=<token de MB Acceso> canjea el token por una sesión.
 * Devuelve { enlaceAcceso, codigoPorCorreo, sesion: { token, email, recordado } | null, error }.
 */
function arranqueIngreso_(p) {
  const r = { enlaceAcceso: '', codigoPorCorreo: false, sesion: null, error: '', url: CONFIG.URL_APP };
  try {
    const cfg = configuracionIngreso_();
    r.enlaceAcceso = cfg.enlaceAcceso;
    r.codigoPorCorreo = cfg.codigoPorCorreo;
    if (p && p.acceso) {
      const s = canjearTokenGoogle_(String(p.acceso), String(p.r || '') === '1', String(p.desde || ''));
      r.sesion = { token: s.token, email: s.email, recordado: s.recordado };
      r.enlaceAcceso = configuracionIngreso_().enlaceAcceso;
    }
  } catch (err) {
    r.error = err && err.esUsuario ? err.message : 'No se pudo verificar el ingreso con Google. Intente nuevamente.';
    if (!(err && err.esUsuario)) console.error('arranqueIngreso_: ' + (err && err.stack ? err.stack : err));
  }
  return r;
}

/** JSON seguro para incrustar dentro de <script> (sin «</script>» ni comentarios HTML). */
function jsonParaHtml_(obj) {
  return JSON.stringify(obj).replace(/</g, '\\u003c').replace(/>/g, '\\u003e').replace(/&/g, '\\u0026')
    .replace(/\u2028/g, '\\u2028').replace(/\u2029/g, '\\u2029');
}

/**
 * Inserta el contenido de otro archivo HTML del proyecto (Index.html: <?!= incluir_('UI_Cartera') ?>).
 * Si el parcial todavía no existe (fase no instalada), devuelve '' y la app sigue funcionando.
 */
function incluir_(nombre) {
  try {
    return HtmlService.createHtmlOutputFromFile(nombre).getContent();
  } catch (e) {
    return '<!-- ' + String(nombre).replace(/[^A-Za-z0-9_]/g, '') + ': todavía no instalado -->';
  }
}

// Todas reciben un único objeto: { token, mes: 'AAAA-MM', idSolicitud, datos, cliente: '2.3' }.
// v2.3: la identidad sale SIEMPRE del token de sesión emitido por el servidor tras verificar a la persona con
// Google («MB Acceso») o, si está habilitado, con un código por correo. Ningún correo escrito autoriza nada.

/**
 * Entrada a la app con una sesión ya emitida (token). Sin token válido responde [SESION] y la página muestra
 * «Ingresar con Google». Con «Ingreso con código por correo» = Sí también acepta { email } y envía el código.
 * Una página vieja (v2.1, sin «cliente») recibe [SESION] y se le pide recargar.
 */
function iniciarSesion(req) {
  return ejecutar_('iniciarSesion', req, 'publico', function (ctx, solicitud) {
    if (!solicitud || solicitud.cliente !== CONFIG.VERSION) {
      throw errorUsuario_(PREFIJO_SESION + 'La aplicación se actualizó a la versión ' + CONFIG.VERSION + '. Recargue la página (tecla F5 o deslizar hacia abajo) e ingrese con Google.');
    }
    const r = conBloqueo_(function () {
      const base = obtenerOCrearHoja_();
      if (base.creada) prepararHojaNueva_(base.ss);
      return base;
    });
    const ss = r.ss;
    const tz = zonaHoraria_(ss);
    const sesion = ctx.token ? leerSesion_(ctx.token) : null;
    if (sesion) {
      const acceso = evaluarAccesoCorreo_(ss, sesion.e, tz);
      return completarIngreso_(ss, tz, sesion.e, ctx.token, sesion, ctx.mes, r, acceso);
    }
    if (solicitud.email && codigoPorCorreoHabilitado_()) {
      const email = normalizarCorreo_(solicitud.email, false);
      const acceso = evaluarAccesoCorreo_(ss, email, tz);
      return solicitarCodigoAcceso_(ss, tz, email, acceso);
    }
    throw errorUsuario_(PREFIJO_SESION + 'Ingrese con el botón «Ingresar con Google».');
  });
}

/** Respaldo (solo si «Ingreso con código por correo» = Sí): valida el código recibido por correo y abre la sesión. */
function verificarCodigo(req) {
  return ejecutar_('verificarCodigo', req, 'publico', function (ctx, solicitud) {
    if (!codigoPorCorreoHabilitado_()) throw errorUsuario_(PREFIJO_SESION + 'El ingreso con código está desactivado. Ingrese con Google.');
    const email = normalizarCorreo_(solicitud.email, false);
    const ss = abrirHoja_();
    const tz = zonaHoraria_(ss);
    const acceso = evaluarAccesoCorreo_(ss, email, tz);
    comprobarCodigoAcceso_(ss, tz, email, solicitud.codigo);
    const nueva = crearSesion_(email, solicitud.recordar === true || solicitud.recordar === 'true', 'CÓDIGO');
    return completarIngreso_(ss, tz, email, nueva.token, nueva.sesion, ctx.mes, { creada: false, avisos: [] }, acceso);
  });
}

/** Cierra la sesión de este dispositivo (revoca su token en el servidor). */
function cerrarSesion(req) {
  return ejecutar_('cerrarSesion', req, 'publico', function (ctx) {
    const s = ctx.token ? leerSesion_(ctx.token) : null;
    if (s) {
      revocarSesion_(ctx.token);
      try {
        const ss = abrirHoja_();
        conBloqueo_(function () { auditarEvento_(ss, s.e, '', 'Salida', 'Sesión cerrada por la usuaria en este dispositivo.', '', '', '', s.m || ''); });
      } catch (e) { console.warn('Auditoría de salida no registrada: ' + e); }
    }
    return { ok: true };
  });
}

  });
}

// ─────────────────────── 4. Ejecución segura (bloqueo, sesión, rol, idempotencia, auditoría) ───────────────────────

/**
 * Envoltorio de TODA llamada desde el navegador.
 *  - 'publico': solo el ingreso (iniciarSesion, verificarCodigo, cerrarSesion).
 *  - 'lectura' / 'escritura': exige sesión válida (token), resuelve el rol y verifica PERMISOS[nombre]
 *    en el servidor. Las escrituras van con bloqueo (30 s), idempotencia por idSolicitud, guardado inmediato
 *    (flush) y auditoría con correo verificado, rol, función, idSolicitud y unidad de negocio.
 *  - v2.3: a una OPERATIVA nunca le llega un campo sensible (costos, márgenes, condiciones de proveedores…):
 *    además de la lista blanca de cada respuesta, se filtra todo el resultado (segunda barrera).
 */
function ejecutar_(nombre, req, modo, fn) {
  try {
    req = (req && typeof req === 'object') ? req : {};
    reiniciarMemoriaEjecucion_();
    const ctx = {
      mes: normalizarMes_(req.mes),
      datos: (req.datos && typeof req.datos === 'object') ? req.datos : {},
      token: String(req.token || '').replace(/[^A-Za-z0-9_-]/g, '').slice(0, 200),
      idSolicitud: String(req.idSolicitud || ''),
    };
    if (modo === 'publico') return fn(ctx, req);

    ctx.ss = abrirHoja_();
    ctx.tz = zonaHoraria_(ctx.ss);
    ctx.usuario = usuarioDeSesion_(ctx.ss, ctx.token, ctx.tz); // lanza [SESION] si no hay sesión válida
    ctx.email = ctx.usuario.email;
    ctx.unidad = unidadSolicitud_(ctx.datos);
    verificarPermiso_(nombre, ctx.usuario);
    const operativa = !esAdministradora_(ctx.usuario);
    if (modo === 'lectura') {
      const r = fn(ctx);
      return operativa ? filtrarSensibles_(r, 0) : r;
    }

    const resultado = conBloqueo_(function () {
      return conIdempotencia_(nombre, ctx, function () {
        iniciarAuditoria_(ctx.usuario, nombre, ctx.idSolicitud, ctx.unidad);
        let r;
        try {
          r = fn(ctx);
        } catch (e) {
          volcarAuditoria_(ctx.ss, ctx.tz, e && e.esUsuario ? '' : ('Operación incompleta: ' + (e && e.message ? e.message : e)));
          throw e;
        }
        volcarAuditoria_(ctx.ss, ctx.tz, '');
        return r;
      });
    });
    return operativa ? filtrarSensibles_(resultado, 0) : resultado;
  } catch (e) {
    throw aErrorCliente_(e, nombre);
  }
}

/** Unidad de negocio declarada en la solicitud (DISTRIBUIDORA / SALÓN); vacía si no corresponde. */
function unidadSolicitud_(datos) {
  try { return unidadValida_(datos && datos.unidad, ''); } catch (e) { return ''; }
}

/**
 * Idempotencia: si la misma solicitud (idSolicitud + operación + datos + usuario) ya se procesó en las
 * últimas 6 horas, devuelve el resultado guardado marcado como «repetida» y no vuelve a escribir.
 */
function conIdempotencia_(nombre, ctx, fn) {
  const cache = CacheService.getScriptCache();
  const clave = claveIdempotencia_(ctx.idSolicitud, nombre + '|' + (ctx.email || ''), ctx.datos);
  if (clave) {
    const previo = cache.get(clave);
    if (previo) {
      const repetido = JSON.parse(previo);
      repetido.repetida = true;
      repetido.datos = ctx.ss ? construirDatos_(ctx.ss, ctx.mes, ctx.tz, ctx.usuario) : null;
      return repetido;
    }
  }
  let resultado = fn();
  if (typeof resultado === 'string') resultado = { mensaje: resultado };
  resultado.ok = true;
  SpreadsheetApp.flush();
  if (clave) {
    try { cache.put(clave, JSON.stringify(resultado), 21600); } catch (e) { console.warn('Idempotencia sin caché: ' + e); }
  }
  if (ctx.ss) resultado.datos = construirDatos_(ctx.ss, ctx.mes, ctx.tz, ctx.usuario);
  return resultado;
}

let bloqueoTomado_ = false;
function conBloqueo_(fn) {
  if (bloqueoTomado_) return fn(); // v2.2: reentrante dentro de la misma ejecución (evita bloqueos anidados)
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(30000)) {
    throw errorUsuario_('El sistema está guardando otra operación. Espere unos segundos e intente nuevamente.');
  }
  bloqueoTomado_ = true;
  try {
    return fn();
  } finally {
    bloqueoTomado_ = false;
    lock.releaseLock();
  }
}

function claveIdempotencia_(id, nombre, datos) {
  const limpio = String(id || '').replace(/[^A-Za-z0-9_-]/g, '').slice(0, 60);
  if (!limpio) return '';
  const bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, nombre + '|' + JSON.stringify(datos));
  let huella = '';
  for (let i = 0; i < bytes.length; i++) huella += ('0' + (bytes[i] & 0xff).toString(16)).slice(-2);
  return 'mb_' + limpio + '_' + huella;
}

function errorUsuario_(mensaje) {
  const e = new Error(mensaje);
  e.esUsuario = true;
  return e;
}

function aErrorCliente_(e, nombre) {
  if (e && e.esUsuario) return new Error(e.message);
  console.error('[' + nombre + '] ' + (e && e.stack ? e.stack : e));
  const detalle = e && e.message ? e.message : String(e);
  if (/permission|permiso|authoriz|autoriz/i.test(detalle)) {
    return new Error('Faltan permisos de Google. Quien instaló el sistema debe abrir el proyecto en script.google.com, ' +
      'elegir la función «autorizar», pulsar «Ejecutar» y aceptar todos los permisos («Seleccionar todo»).');
  }
  return new Error('No se pudo completar la operación. Intente nuevamente en unos segundos. Detalle técnico: ' + detalle);
}

// ─────────────────────── 5. Hoja de cálculo: creación, recuperación y permisos ───────────────────────

function obtenerOCrearHoja_() {
  if (MODO_PRUEBA_ && MODO_PRUEBA_.libro) return { ss: MODO_PRUEBA_.libro, creada: false, avisos: [] };
  const props = PropertiesService.getScriptProperties();
  const avisos = [];
  const idGuardado = props.getProperty(CONFIG.CLAVE_ID_HOJA);
  let archivo = null;

  if (idGuardado) {
    try {
      archivo = DriveApp.getFileById(idGuardado);
      if (archivo.isTrashed()) {
        archivo.setTrashed(false);
        avisos.push('La hoja de cálculo estaba en la papelera de Google Drive y se restauró automáticamente.');
      }
    } catch (e) {
      archivo = null; // Eliminada definitivamente o sin acceso.
    }
  }
  if (!archivo) archivo = buscarHojaPorNombre_();

  let ss;
  let creada = false;
  if (archivo) {
    ss = SpreadsheetApp.openById(archivo.getId());
  } else {
    ss = SpreadsheetApp.create(CONFIG.NOMBRE_HOJA);
    creada = true;
    if (idGuardado) avisos.push('No se encontró la hoja anterior (fue eliminada definitivamente). Se creó una hoja nueva.');
  }
  props.setProperty(CONFIG.CLAVE_ID_HOJA, ss.getId());
  return { ss: ss, creada: creada, avisos: avisos };
}

/** Si el ID se perdió (por ejemplo, al copiar el proyecto), reutiliza la hoja existente del propietario. */
function buscarHojaPorNombre_() {
  const yo = correoPropietario_();
  let elegido = null;
  try {
    const it = DriveApp.getFilesByName(CONFIG.NOMBRE_HOJA);
    while (it.hasNext()) {
      const f = it.next();
      try {
        if (f.isTrashed() || f.getMimeType() !== MimeType.GOOGLE_SHEETS) continue;
        const dueno = f.getOwner();
        if (yo && dueno && String(dueno.getEmail()).toLowerCase() !== yo) continue;
        if (!elegido || f.getLastUpdated() > elegido.getLastUpdated()) elegido = f;
      } catch (e) { /* se ignora ese archivo */ }
    }
  } catch (e) { /* sin resultados */ }
  return elegido;
}

}

/**
 * Comparte la hoja con EDICIÓN SOLO a administradoras, sin correo de invitación (servicio avanzado de Drive).
 * v2.3: ya NO se llama al ingresar (antes compartía el libro con cada correo que entraba). Devuelve
 * { correo: 'Propietario' | 'Editor' | 'Error: …' }.
 */
function asegurarAcceso_(ss, correos) {
  const resultado = {};
  const archivo = DriveApp.getFileById(ss.getId());
  let dueno = '';
  try {
    const d = archivo.getOwner();
    dueno = d ? String(d.getEmail()).toLowerCase() : '';
  } catch (e) { /* sin dato */ }
  correos.forEach(function (correo) {
    correo = String(correo || '').toLowerCase().trim();
    if (!correo) return;
    if (correo === dueno || correo === correoPropietario_()) { resultado[correo] = 'Propietario'; return; }
    const u = usuarioPorCorreo_(ss, correo);
    if (u.rol !== ROLES.ADMIN || !u.activo) { resultado[correo] = 'Error: solo se comparte con administradoras activas.'; return; }
    const r = compartirSinAviso_(ss.getId(), correo, 'writer');
    resultado[correo] = r === 'OK' || r === 'YA TENÍA' ? 'Editor' : 'Error: ' + r.replace(/^ERROR: /, '');
  });
  return resultado;
}

function registrarIngreso_(tabla, existente, email, acceso, tz, extra) {
  const ahora = new Date();
  extra = extra || {};
  if (existente) {
    const cambios = {
      ultimo: ahora,
      ingresos: (existente.ingresos || 0) + 1,
      acceso: acceso || existente.acceso,
    };
    if (extra.verificado) cambios.verificado = ahora;
    if (extra.metodo) cambios.metodo = extra.metodo;
    if (extra.rol && !existente.rol) cambios.rol = extra.rol;
    actualizarCeldas_(tabla, existente._fila, cambios, tz);
  } else {
    agregarFilas_(tabla, [{
      correo: email, activo: 'Sí', acceso: acceso || '', alta: ahora, ultimo: ahora, ingresos: 1,
      rol: extra.rol || ROLES.OPERATIVO, verificado: extra.verificado ? ahora : '', metodo: extra.metodo || '', notas: extra.notas || '',
    }], tz);
  }
}

/** v2.3 (R6.3): si falta la pestaña, la fila o el valor, el registro queda CERRADO (la v2.1 lo dejaba abierto). */
function registroAbierto_(ss) {
  const hoja = ss.getSheetByName(HOJA_CONFIG);
  if (!hoja || hoja.getLastRow() < 2) return false;
  const valores = hoja.getRange(1, 1, hoja.getLastRow(), 2).getValues();
  for (let i = 0; i < valores.length; i++) {
    if (String(valores[i][0]).trim() === PARAM_REGISTRO) {
      return /^abierto\b/i.test(String(valores[i][1]).trim());
    }
  }
  return false;
}

/**
 * Crea las hojas que falten y repara títulos ausentes (columnas nuevas SOLO al final). Es seguro ejecutarla
 * siempre: no renombra ni borra hojas, títulos ni datos. Ubica cada hoja nueva junto a su referencia
 * («despuesDe») y agrega las filas nuevas de «Configuración». Devuelve los nombres de las hojas creadas.
 */
function asegurarEstructura_(ss) {
  const existentes = {};
  ss.getSheets().forEach(function (h) { existentes[h.getName()] = h; });
  if (!existentes[HOJA_LEAME]) escribirLeame_(ss.insertSheet(HOJA_LEAME, 0), ss);
  const creadas = [];
  Object.keys(ESQUEMA).forEach(function (clave) {
    const def = ESQUEMA[clave];
    let hoja = existentes[def.hoja];
    if (!hoja) {
      hoja = ss.insertSheet(def.hoja, posicionNuevaHoja_(ss, def));
      formatearHoja_(hoja, def);
      existentes[def.hoja] = hoja;
      creadas.push(def.hoja);
    } else {
      repararEncabezados_(hoja, def);
    }
  });
  if (!existentes[HOJA_CONFIG]) crearConfiguracion_(ss.insertSheet(HOJA_CONFIG, ss.getSheets().length));
  asegurarFilasConfiguracion_(ss);
  return creadas;
}

/**
 * Índice (base 0) para insertar una hoja nueva justo después de su hoja de referencia (def.despuesDe).
 * Si la referencia aún no existe, sigue la cadena de referencias hasta encontrar una.
 */
function posicionNuevaHoja_(ss, def) {
  let actual = def;
  for (let i = 0; i < 30 && actual && actual.despuesDe && ESQUEMA[actual.despuesDe]; i++) {
    const ref = ss.getSheetByName(ESQUEMA[actual.despuesDe].hoja);
    if (ref) return ref.getIndex();
    actual = ESQUEMA[actual.despuesDe];
  }
  return ss.getSheets().length;
}

function formatearHoja_(hoja, def) {
  if (hoja.getMaxRows() > 300 && hoja.getLastRow() <= 1) hoja.deleteRows(301, hoja.getMaxRows() - 300); // hojas nuevas livianas
  escribirEncabezados_(hoja, def.columnas, 1);
  hoja.setFrozenRows(1);
  hoja.setRowHeight(1, 34);
  hoja.setTabColor(COLORES[def.grupo] || COLORES.sistema);
  const n = def.columnas.length;
  const proteccion = def.grupo === 'catalogo' ? hoja.getRange(1, 1, 1, n).protect() : hoja.protect();
  proteccion.setDescription(def.grupo === 'catalogo'
    ? 'Títulos del sistema: no los modifique.'
    : 'Registro gestionado por la aplicación. Para corregir, use «Anular» en la aplicación.');
  proteccion.setWarningOnly(true);
}

/**
 * Escribe títulos, anchos, formatos y (v2.3) validaciones de lista con AVISO: lo ya cargado que no esté en la
 * lista queda marcado, pero no se rechaza ni se pierde.
 */
function escribirEncabezados_(hoja, columnas, desde) {
  const hasta = desde + columnas.length - 1;
  if (hoja.getMaxColumns() < hasta) hoja.insertColumnsAfter(hoja.getMaxColumns(), hasta - hoja.getMaxColumns());
  if (hoja.getMaxRows() < 2) hoja.insertRowsAfter(hoja.getMaxRows(), 200);
  hoja.getRange(1, desde, 1, columnas.length).setValues([columnas.map(function (c) { return c[1]; })]);
  estiloEncabezado_(hoja.getRange(1, desde, 1, columnas.length));
  const filas = hoja.getMaxRows() - 1;
  columnas.forEach(function (c, i) {
    hoja.setColumnWidth(desde + i, c[3] || 120);
    const rango = hoja.getRange(2, desde + i, filas, 1);
    if (FORMATOS[c[2]]) rango.setNumberFormat(FORMATOS[c[2]]);
    if (c[2] === 'texto' && /hora/i.test(c[1]) && !/fecha/i.test(c[1])) rango.setNumberFormat('@');
    const lista = c[0] === 'activo' ? LISTAS.SI_NO : (c[4] && c[4].lista ? c[4].lista : null);
    if (lista) rango.setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(lista, true).setAllowInvalid(true).build());
  });
}

function repararEncabezados_(hoja, def) {
  const ultima = hoja.getLastColumn();
  const actuales = ultima > 0
    ? hoja.getRange(1, 1, 1, ultima).getValues()[0].map(function (v) { return String(v).trim(); })
    : [];
  const vacia = actuales.every(function (v) { return v === ''; }) && hoja.getLastRow() <= 1;
  if (vacia) {
    formatearHoja_(hoja, def);
    return;
  }
  const alFinal = [];
  def.columnas.forEach(function (c, i) {
    if (actuales.indexOf(c[1]) >= 0) return;
    if (i < actuales.length && actuales[i] === '' && i < ultima) {
      // El título se borró pero la columna sigue en su lugar original: se restaura ahí y los datos se conservan.
      hoja.getRange(1, i + 1).setValue(c[1]);
      estiloEncabezado_(hoja.getRange(1, i + 1));
      actuales[i] = c[1];
    } else {
      alFinal.push(c);
    }
  });
  if (alFinal.length) escribirEncabezados_(hoja, alFinal, ultima + 1);
}

// ─────────────────────── 7. Lectura y escritura de tablas ───────────────────────

/** Lee una hoja del esquema y devuelve sus filas como objetos (fechas como texto 'AAAA-MM-DD'). */
function leerTabla_(ss, clave, tz) {
  const def = ESQUEMA[clave];
  const hoja = ss.getSheetByName(def.hoja);
  if (!hoja) {
    throw errorUsuario_(PREFIJO_SESION + 'Falta la hoja «' + def.hoja + '». Avise a la administración: al ingresar una administradora el sistema la repara.');
  }
  const valores = hoja.getDataRange().getValues();
  const encabezados = (valores[0] || []).map(function (v) { return String(v).trim(); });
  const idx = {};
  const tipos = {};
  def.columnas.forEach(function (c) {
    const i = encabezados.indexOf(c[1]);
    if (i < 0) {
      throw errorUsuario_(PREFIJO_SESION + 'Falta la columna «' + c[1] + '» en la hoja «' + def.hoja + '». Avise a la administración: al ingresar una administradora el sistema la repara.');
    }
    idx[c[0]] = i;
    tipos[c[0]] = c[2];
  });
  const filas = [];
  for (let r = 1; r < valores.length; r++) {
    const fila = valores[r];
    if (fila.every(function (v) { return v === '' || v === null; })) continue;
    const obj = { _fila: r + 1 };
    def.columnas.forEach(function (c) { obj[c[0]] = deCelda_(fila[idx[c[0]]], c[2], tz); });
    filas.push(obj);
  }
  return { hoja: hoja, def: def, idx: idx, tipos: tipos, ancho: Math.max(encabezados.length, 1), filas: filas };
}

function agregarFilas_(tabla, objetos, tz) {
  if (!objetos.length) return;
  const hoja = tabla.hoja;
  const claves = Object.keys(tabla.idx);
  const filas = objetos.map(function (o) {
    const fila = new Array(tabla.ancho).fill('');
    claves.forEach(function (k) {
      if (Object.prototype.hasOwnProperty.call(o, k)) fila[tabla.idx[k]] = aCelda_(o[k], tabla.tipos[k], tz);
    });
    return fila;
  });
  const inicio = hoja.getLastRow() + 1;
  const fin = inicio + filas.length - 1;
  const max = hoja.getMaxRows();
  if (fin > max) hoja.insertRowsAfter(max, fin - max + 200);
  hoja.getRange(inicio, 1, filas.length, tabla.ancho).setValues(filas);
  claves.forEach(function (k) {
    const formato = FORMATOS[tabla.tipos[k]];
    if (formato) hoja.getRange(inicio, tabla.idx[k] + 1, filas.length, 1).setNumberFormat(formato);
  });
  // v2.2: las filas nuevas quedan también en memoria (para cálculos posteriores de la misma operación) y en la auditoría.
  if (tabla.filas) {
    objetos.forEach(function (o, i) {
      const obj = { _fila: inicio + i };
      claves.forEach(function (k) { obj[k] = deCelda_(filas[i][tabla.idx[k]], tabla.tipos[k], tz); });
      tabla.filas.push(obj);
    });
  }
  registrarAltaAuditoria_(tabla, objetos, tz);
}

function actualizarCeldas_(tabla, fila, cambios, tz) {
  const enMemoria = tabla.filas ? tabla.filas.find(function (f) { return f._fila === fila; }) : null;
  Object.keys(cambios).forEach(function (k) {
    if (tabla.idx[k] === undefined) return;
    const tipo = tabla.tipos[k];
    const celda = tabla.hoja.getRange(fila, tabla.idx[k] + 1);
    const valor = aCelda_(cambios[k], tipo, tz);
    celda.setValue(valor);
    if (tipo === 'fecha' || tipo === 'fechahora') celda.setNumberFormat(FORMATOS[tipo]);
    // v2.2: auditoría (valor anterior → nuevo) y copia en memoria actualizada.
    if (enMemoria) {
      const nuevo = deCelda_(valor, tipo, tz);
      registrarCambioAuditoria_(tabla, enMemoria, k, enMemoria[k], nuevo);
      enMemoria[k] = nuevo;
    }
  });
}

/**
 * v2.2: escribe de una sola vez una columna calculada (por ejemplo «Saldo (automático)»).
 * valores: { númeroDeFila: valor }. Las filas que no figuran conservan su contenido.
 */
function escribirColumna_(tabla, clave, valores, tz) {
  if (tabla.idx[clave] === undefined) return;
  const ultima = tabla.hoja.getLastRow();
  if (ultima < 2) return;
  const rango = tabla.hoja.getRange(2, tabla.idx[clave] + 1, ultima - 1, 1);
  const actuales = rango.getValues();
  let cambio = false;
  for (let r = 2; r <= ultima; r++) {
    if (!Object.prototype.hasOwnProperty.call(valores, r)) continue;
    const v = aCelda_(valores[r], tabla.tipos[clave], tz);
    const previo = actuales[r - 2][0];
    const igual = (v instanceof Date && previo instanceof Date) ? v.getTime() === previo.getTime()
      : String(v).replace(/^'/, '') === String(previo);
    if (!igual) { actuales[r - 2][0] = v; cambio = true; }
  }
  if (cambio) rango.setValues(actuales);
  if (tabla.filas) {
    tabla.filas.forEach(function (f) {
      if (Object.prototype.hasOwnProperty.call(valores, f._fila)) f[clave] = deCelda_(aCelda_(valores[f._fila], tabla.tipos[clave], tz), tabla.tipos[clave], tz);
    });
  }
}

/**
 * v2.3: cambios en muchas filas con una escritura por columna (lotes: migraciones, recálculos, backfills).
 * cambios: [{ fila, valores: { clave: valor } }]. Audita cada campo que cambia (anterior → nuevo).
 * Devuelve la cantidad de filas con cambios.
 */
function actualizarFilas_(tabla, cambios, tz) {
  if (!cambios || !cambios.length) return 0;
  const enMemoria = {};
  if (tabla.filas) tabla.filas.forEach(function (f) { enMemoria[f._fila] = f; });
  const porClave = {};
  let n = 0;
  cambios.forEach(function (c) {
    const f = enMemoria[c.fila];
    let alguno = false;
    Object.keys(c.valores).forEach(function (k) {
      if (tabla.idx[k] === undefined) return;
      const nuevo = deCelda_(aCelda_(c.valores[k], tabla.tipos[k], tz), tabla.tipos[k], tz);
      if (f && textoAuditoria_(f[k]) === textoAuditoria_(nuevo)) return;
      if (f) registrarCambioAuditoria_(tabla, f, k, f[k], nuevo);
      (porClave[k] = porClave[k] || {})[c.fila] = c.valores[k];
      alguno = true;
    });
    if (alguno) n++;
  });
  Object.keys(porClave).forEach(function (k) { escribirColumna_(tabla, k, porClave[k], tz); });
  return n;
}

function deCelda_(valor, tipo, tz) {
  switch (tipo) {
    case 'fecha': return fechaATexto_(valor, tz);
    case 'fechahora': return fechaHoraATexto_(valor, tz);
    case 'gs':
    case 'entero': { const n = entero_(valor); return isNaN(n) ? 0 : n; }
    case 'decimal': { const n = decimal_(valor); return isNaN(n) ? 0 : n; }
    case 'gsopc': { // v2.2: vacío se conserva como '' (no es 0)
      if (valor === '' || valor === null || valor === undefined) return '';
      const n = entero_(valor);
      return isNaN(n) ? '' : n;
    }
    case 'libre': // v2.2: número o texto tal cual
      if (valor instanceof Date) return fechaATexto_(valor, tz);
      if (typeof valor === 'number') return valor;
      return String(valor === null || valor === undefined ? '' : valor).trim();
    default:
      if (valor instanceof Date) return fechaATexto_(valor, tz);
      return String(valor === null || valor === undefined ? '' : valor).trim();
  }
}

function aCelda_(valor, tipo, tz) {
  if (valor === null || valor === undefined || valor === '') return '';
  switch (tipo) {
    case 'fecha': return valor instanceof Date ? valor : textoAFecha_(String(valor), tz);
    case 'fechahora': return valor instanceof Date ? valor : (/^\d{4}-\d{2}-\d{2}( \d{2}:\d{2})?$/.test(String(valor)) ? textoAFechaHora_(String(valor), tz) : '');
    case 'gs':
    case 'entero': return Math.round(Number(valor)) || 0;
    case 'gsopc': { const n = Math.round(Number(valor)); return isNaN(n) ? '' : n; }
    case 'decimal': return Number(valor) || 0;
    case 'libre': return typeof valor === 'number' ? valor : celdaTexto_(String(valor));
    default: return celdaTexto_(String(valor));
  }
}

/** Evita que Sheets convierta texto en número, fecha o fórmula (ej.: "0981…", "+595…", "=…"). */
function celdaTexto_(s) {
  s = String(s === null || s === undefined ? '' : s);
  if (!s) return s;
  if (/^[=+\-@0-9.,'#$(]/.test(s) || /^(true|false|verdadero|falso)$/i.test(s)) return "'" + s;
  return s;
}

/** Escribe el stock calculado en la columna «Stock actual (automático)» de Productos. */
function escribirStock_(tabla, stock) {
  if (!tabla.filas.length) return;
  const porFila = {};
  let ultima = 2;
  tabla.filas.forEach(function (p) {
    porFila[p._fila] = stock[clave_(p.codigo)] || 0;
    if (p._fila > ultima) ultima = p._fila;
  });
  const valores = [];
  for (let r = 2; r <= ultima; r++) {
    valores.push([Object.prototype.hasOwnProperty.call(porFila, r) ? porFila[r] : '']);
  }
  tabla.hoja.getRange(2, tabla.idx.stock + 1, valores.length, 1).setValues(valores);
}

function calcularStock_(movimientos) {
  const stock = {};
  movimientos.forEach(function (m) {
    const k = clave_(m.codigo);
    if (k) stock[k] = (stock[k] || 0) + (Number(m.cantidad) || 0);
  });
  return stock;
}

function guardarProducto_(ctx) {
  const d = ctx.datos;
  const tz = ctx.tz;
  const tabla = leerTabla_(ctx.ss, 'productos', tz);
  const datos = {
    nombre: requerirTexto_(d.nombre, 'Producto', 120),
    marca: texto_(d.marca, 60),
    categoria: texto_(d.categoria, 60),
    presentacion: texto_(d.presentacion, 40),
    costo: enteroOpcional_(d.costo, 'Costo unitario', LIMITES.PRECIO_UNITARIO),
    precio: requerirEntero_(d.precio, 'Precio a peluquerías', 0, LIMITES.PRECIO_UNITARIO),
    stockMinimo: enteroOpcional_(d.stockMinimo, 'Stock mínimo', LIMITES.CANTIDAD),
    activo: activoDe_(d.activo),
    actualizado: new Date(),
    usuario: ctx.email,
  };
  // v2.3: Tasa IVA, Línea y Código proveedor solo se tocan si la pantalla los envía (las pantallas viejas no los borran).
  if (Object.prototype.hasOwnProperty.call(d, 'iva')) datos.iva = String(d.iva) === '' ? '' : textoTasa_(tasaIva_(d.iva, null));
  if (Object.prototype.hasOwnProperty.call(d, 'linea')) {
    const linea = texto_(d.linea, 60).toUpperCase();
    if (linea && LISTAS.LINEAS.indexOf(linea) < 0) throw errorUsuario_('Línea / centro de costo no válida. Opciones: ' + LISTAS.LINEAS.join(', ') + '.');
    datos.linea = linea;
  }
  if (Object.prototype.hasOwnProperty.call(d, 'proveedor')) {
    const prov = clave_(d.proveedor);
    if (prov) {
      const provs = leerTabla_(ctx.ss, 'proveedores', tz).filas;
      if (!buscarPorCodigo_(provs, prov)) throw errorUsuario_('El proveedor ' + prov + ' no existe en «Proveedores».');
    }
    datos.proveedor = prov;
  }

  const original = clave_(d.codigoOriginal);
  let existente = null;
  if (original) {
    existente = buscarPorCodigo_(tabla.filas, original);
    if (!existente) throw errorUsuario_('El producto ' + original + ' no existe. Actualice los datos e intente nuevamente.');
  }
  const duplicado = tabla.filas.find(function (p) {
    return p !== existente && normalizar_(p.nombre) === normalizar_(datos.nombre) && normalizar_(p.marca) === normalizar_(datos.marca);
  });
  if (duplicado) {
    throw errorUsuario_('Ya existe «' + duplicado.nombre + '»' + (duplicado.marca ? ' de ' + duplicado.marca : '') + ' (código ' + duplicado.codigo + ').');
  }

  if (existente) {
    actualizarCeldas_(tabla, existente._fila, datos, tz);
    return { mensaje: 'Producto ' + existente.codigo + ' actualizado.', codigo: existente.codigo };
  }

  let codigo = clave_(d.codigo);
  if (codigo) {
    if (!/^[A-Z0-9][A-Z0-9._\-\/]{0,29}$/.test(codigo)) {
      throw errorUsuario_('El código solo admite letras, números, guiones, puntos o barras (máximo 30 caracteres).');
    }
    if (buscarPorCodigo_(tabla.filas, codigo)) throw errorUsuario_('El código ' + codigo + ' ya está en uso.');
  } else {
    codigo = siguienteCodigo_(tabla.filas.map(function (p) { return p.codigo; }), 'P-', 4);
  }
  const stockInicial = enteroOpcional_(d.stockInicial, 'Stock inicial', LIMITES.CANTIDAD);
  datos.codigo = codigo;
  datos.stock = stockInicial;
  if (datos.iva === undefined) datos.iva = '10';
  if (datos.linea === undefined) datos.linea = lineaPropuesta_(datos.categoria, false);
  agregarFilas_(tabla, [datos], tz);

  if (stockInicial > 0) {
    const movs = leerTabla_(ctx.ss, 'movimientos', tz);
    agregarFilas_(movs, [{
      id: siguienteCodigo_(movs.filas.map(function (m) { return m.id; }), 'M-', 6),
      fecha: hoy_(tz), tipo: 'Stock inicial', codigo: codigo, producto: datos.nombre,
      cantidad: stockInicial, costoUnitario: datos.costo, referencia: codigo,
      nota: 'Alta del producto', usuario: ctx.email, registrado: new Date(), unidad: 'DISTRIBUIDORA',
    }], tz);
  }
  return {
    mensaje: 'Producto ' + codigo + ' creado' + (stockInicial > 0 ? ' con ' + miles_(stockInicial) + ' unidades en stock.' : '.'),
    codigo: codigo,
  };
}

/**
 * v2.3 (R4.3): «Uso en salón» es una transferencia interna de la distribuidora al salón, valorizada con el
 * «Precio de transferencia» de «Configuración» (por defecto, el costo MB).
 */
function precioTransferencia_(ss, producto) {
  const modo = String(leerConfig_(ss, CFG.TRANSFERENCIA) || 'COSTO MB').toUpperCase();
  if (/LISTA|PRECIO/.test(modo)) return Number(producto.precio) || 0;
  return Number(producto.costo) || 0;
}

function registrarMovimiento_(ctx) {
  const d = ctx.datos;
  const tz = ctx.tz;
  const tipo = TIPOS_MOVIMIENTO[d.tipo];
  if (!tipo) throw errorUsuario_('Seleccione el tipo de movimiento.');
  const productos = leerTabla_(ctx.ss, 'productos', tz);
  const producto = buscarPorCodigo_(productos.filas, d.codigo);
  if (!producto) throw errorUsuario_('Seleccione un producto válido.');
  if (!esActivo_(producto.activo)) {
    throw errorUsuario_('«' + producto.nombre + '» está inactivo. Actívelo (Editar) antes de registrar movimientos.');
  }
  const cantidad = requerirEntero_(d.cantidad, 'Cantidad', 1, LIMITES.CANTIDAD);
  const fecha = validarFecha_(d.fecha, tz, 'el movimiento');
  const nota = texto_(d.nota, 300);
  if (tipo.notaObligatoria && nota.length < 3) throw errorUsuario_('Escriba el motivo del ajuste en «Nota».');
  // Solo una administradora puede informar el costo de una compra (las operativas no ven costos).
  const puedeCosto = esAdministradora_(ctx.usuario);
  const costoNuevo = d.tipo === 'ingreso' && puedeCosto ? enteroOpcional_(d.costoUnitario, 'Costo unitario', LIMITES.PRECIO_UNITARIO, true) : null;

  const movs = leerTabla_(ctx.ss, 'movimientos', tz);
  const stock = calcularStock_(movs.filas);
  const k = clave_(producto.codigo);
  const actual = stock[k] || 0;
  if (tipo.signo < 0 && cantidad > actual) {
    throw errorUsuario_('Stock insuficiente de «' + producto.nombre + '»: disponible ' + miles_(actual) + ', solicitado ' + miles_(cantidad) + '.');
  }
  const id = siguienteCodigo_(movs.filas.map(function (m) { return m.id; }), 'M-', 6);
  const usoSalon = d.tipo === 'uso';
  agregarFilas_(movs, [{
    id: id, fecha: fecha, tipo: tipo.etiqueta, codigo: producto.codigo, producto: producto.nombre,
    cantidad: tipo.signo * cantidad, costoUnitario: costoNuevo !== null ? costoNuevo : producto.costo,
    referencia: '', nota: nota, usuario: ctx.email, registrado: new Date(),
    unidad: usoSalon ? 'SALÓN' : 'DISTRIBUIDORA',
    precioTransferencia: usoSalon ? precioTransferencia_(ctx.ss, producto) : '',
  }], tz);
  stock[k] = actual + tipo.signo * cantidad;
  escribirStock_(productos, stock);

  let extra = '';
  if (costoNuevo !== null && costoNuevo > 0 && costoNuevo !== producto.costo) {
    actualizarCeldas_(productos, producto._fila, { costo: costoNuevo, actualizado: new Date(), usuario: ctx.email }, tz);
    extra = ' Costo unitario actualizado a Gs. ' + miles_(costoNuevo) + '.';
  }
  return 'Movimiento ' + id + ' registrado. Stock de «' + producto.nombre + '»: ' + miles_(stock[k]) + '.' + extra;
}

/**
 * Alta o edición de una peluquería / clienta de la distribuidora. Con el módulo de clientes (Clientes.gs, R1)
 * se aplican los obligatorios por tipo; sin él, se conserva el comportamiento vigente con RUC y celular validados.
 */
function guardarPeluqueria_(ctx) {
  if (typeof guardarCliente_ === 'function') return guardarCliente_(ctx, 'peluquerias');
  const d = ctx.datos;
  const tabla = leerTabla_(ctx.ss, 'peluquerias', ctx.tz);
  const original = clave_(d.codigoOriginal);
  const actual = original ? buscarPorCodigo_(tabla.filas, original) : null;
  // Al editar, un campo que la pantalla no envía conserva su valor (no se borra por omisión).
  const tiene = function (k) { return !actual || Object.prototype.hasOwnProperty.call(d, k); };
  const datos = {
    nombre: requerirTexto_(d.nombre, 'Peluquería', 100),
    contacto: tiene('contacto') ? texto_(d.contacto, 80) : actual.contacto,
    telefono: tiene('telefono') ? texto_(d.telefono, 40) : actual.telefono,
    ciudad: tiene('ciudad') ? texto_(d.ciudad, 60) : actual.ciudad,
    direccion: tiene('direccion') ? texto_(d.direccion, 150) : actual.direccion,
    ruc: tiene('ruc') ? validarRuc_(d.ruc, { validarDv: configSi_(ctx.ss, CFG.RUC_DV) }) : actual.ruc,
    activo: tiene('activo') ? activoDe_(d.activo) : actual.activo,
    actualizado: new Date(),
    usuario: ctx.email,
  };
  const celular = normalizarCelular_(tiene('celular') && d.celular ? d.celular : datos.telefono);
  if (celular) datos.celular = celular;
  if (!original) datos.alta = hoy_(ctx.tz);
  if (Object.prototype.hasOwnProperty.call(d, 'categoria') && texto_(d.categoria, 30) !== (actual ? actual.categoria : '')) {
    exigirAdministradora_(ctx, 'asignar o cambiar la categoría de crédito');
    datos.categoria = validarCategoria_(ctx.ss, ctx.tz, d.categoria);
  }
  const r = guardarCatalogo_(tabla, d.codigoOriginal, datos, 'C-', 4, 'La peluquería', ctx.tz);
  if (typeof recalcularCartera_ === 'function') recalcularCartera_(ctx.ss, ctx.tz);
  return r;
}

/**
 * Valida lo común a toda venta de productos (fecha, cliente, productos, stock) sin escribir nada.
 * Devuelve las tablas leídas y las líneas listas para escribir (con tasa de IVA, línea y precio de lista).
 */
function prepararVenta_(ctx) {
  const d = ctx.datos;
  const tz = ctx.tz;
  const fecha = validarFecha_(d.fecha, tz, 'la venta');
  const peluquerias = leerTabla_(ctx.ss, 'peluquerias', tz);
  let cliente = buscarPorCodigo_(peluquerias.filas, d.codigoCliente);
  if (!cliente && typeof buscarClienteEnBase_ === 'function') cliente = buscarClienteEnBase_(ctx.ss, tz, d.codigoCliente);
  if (!cliente) throw errorUsuario_('Seleccione el cliente.');
  if (!esActivo_(cliente.activo)) throw errorUsuario_('El cliente «' + cliente.nombre + '» está inactivo.');
  const items = Array.isArray(d.items) ? d.items : [];
  if (!items.length) throw errorUsuario_('Agregue al menos un producto a la venta.');
  if (items.length > LIMITES.ITEMS_POR_VENTA) {
    throw errorUsuario_('Una venta admite hasta ' + LIMITES.ITEMS_POR_VENTA + ' líneas. Divídala en dos ventas.');
  }
  const P = leerParametros_(ctx.ss, tz);
  const productos = leerTabla_(ctx.ss, 'productos', tz);
  const movs = leerTabla_(ctx.ss, 'movimientos', tz);
  const stock = calcularStock_(movs.filas);
  const pedido = {};
  const lineas = items.map(function (it, i) {
    const p = buscarPorCodigo_(productos.filas, it && it.codigo);
    if (!p) throw errorUsuario_('El producto de la línea ' + (i + 1) + ' no existe.');
    if (!esActivo_(p.activo)) throw errorUsuario_('«' + p.nombre + '» está inactivo.');
    const cantidad = requerirEntero_(it.cantidad, 'Cantidad de «' + p.nombre + '»', 1, LIMITES.CANTIDAD);
    const precio = requerirEntero_(it.precio, 'Precio de «' + p.nombre + '»', 0, LIMITES.PRECIO_UNITARIO);
    const k = clave_(p.codigo);
    pedido[k] = (pedido[k] || 0) + cantidad;
    const tasa = tasaIva_(p.iva, P);
    return {
      p: p, k: k, cantidad: cantidad, precio: precio, subtotal: cantidad * precio, tasa: tasa,
      lista: Number(p.precio) || 0, linea: p.linea || lineaPropuesta_(p.categoria, false),
    };
  });
  Object.keys(pedido).forEach(function (k) {
    const disponible = stock[k] || 0;
    if (pedido[k] > disponible) {
      const p = buscarPorCodigo_(productos.filas, k);
      throw errorUsuario_('Stock insuficiente de «' + p.nombre + '»: disponible ' + miles_(disponible) + ', en la venta ' + miles_(pedido[k]) + '.');
    }
  });
  const subtotal = lineas.reduce(function (t, l) { return t + l.subtotal; }, 0);
  return {
    fecha: fecha, cliente: cliente, peluquerias: peluquerias, productos: productos, movs: movs, stock: stock,
    lineas: lineas, subtotal: subtotal, nota: texto_(d.nota, 300), P: P,
  };
}

/**
 * Escribe la venta (una fila por línea, mismo N° de venta), descuenta el stock con el mecanismo existente y agrega
 * líneas extra sin stock (recargo financiero «REC-FIN»). op: { condicion, nroCredito, autorizadoPor, motivoAutorizacion,
 * ticket, unidad, extras, origen, id }.
 */
function escribirVenta_(ctx, prep, op) {
  const tz = ctx.tz;
  const ventas = leerTabla_(ctx.ss, 'ventas', tz);
  const id = op.id || siguienteCodigo_(ventas.filas.map(function (v) { return v.id; }), 'V-', 5);
  const ahora = new Date();
  const contado = op.condicion === 'Contado';
  const unidad = op.unidad || 'DISTRIBUIDORA';
  const comunes = {
    id: id, fecha: prep.fecha, codigoCliente: prep.cliente.codigo, cliente: prep.cliente.nombre,
    condicion: op.condicion, estado: contado ? 'Cobrada' : 'Pendiente', fechaCobro: contado ? prep.fecha : '',
    nota: prep.nota, usuario: ctx.email, registrado: ahora,
    ticket: op.ticket || '', unidad: unidad, origen: op.origen || 'NUEVO', nroCredito: op.nroCredito || '',
    autorizadoPor: op.autorizadoPor || '', motivoAutorizacion: op.motivoAutorizacion || '',
  };
  let total = 0;
  const filasVenta = prep.lineas.map(function (l) {
    total += l.subtotal;
    return Object.assign({}, comunes, {
      codigo: l.p.codigo, producto: l.p.nombre, cantidad: l.cantidad, precio: l.precio, subtotal: l.subtotal,
      costoUnitario: l.p.costo, precioLista: l.lista || '', descuento: Math.round(descuentoLinea_(l.precio, l.lista) * 10000) / 100,
      iva: textoTasa_(l.tasa), ivaMonto: ivaIncluido_(l.subtotal, l.tasa), linea: l.linea,
      codigoProfesional: l.codigoProfesional || '', profesional: l.profesional || '', porcentaje: l.porcentaje || '', comision: l.comision || '',
    });
  });
  (op.extras || []).forEach(function (x) {
    total += x.subtotal;
    filasVenta.push(Object.assign({}, comunes, {
      codigo: x.codigo, producto: x.producto, cantidad: x.cantidad, precio: x.precio, subtotal: x.subtotal,
      costoUnitario: 0, precioLista: x.precio, descuento: 0, iva: textoTasa_(x.tasa), ivaMonto: ivaIncluido_(x.subtotal, x.tasa), linea: x.linea || '',
    }));
  });
  let siguiente = siguienteNumero_(prep.movs.filas.map(function (m) { return m.id; }), 'M-');
  const filasMov = prep.lineas.map(function (l) {
    prep.stock[l.k] = (prep.stock[l.k] || 0) - l.cantidad;
    return {
      id: formatoCodigo_('M-', siguiente++, 6), fecha: prep.fecha, tipo: 'Venta', codigo: l.p.codigo,
      producto: l.p.nombre, cantidad: -l.cantidad, costoUnitario: l.p.costo, referencia: id,
      nota: prep.cliente.nombre, usuario: ctx.email, registrado: ahora, unidad: unidad, ticket: op.ticket || '',
    };
  });
  agregarFilas_(ventas, filasVenta, tz);
  agregarFilas_(prep.movs, filasMov, tz);
  escribirStock_(prep.productos, prep.stock);
  return { id: id, total: total, filas: filasVenta.length };
}

/**
 * Venta de productos (compatibilidad con la pantalla vigente). Con el módulo de tickets (Tickets.gs, R4) toda venta
 * crea su ticket; con el de cartera (Cartera.gs, R0) la venta a crédito exige plan y política.
 */
function registrarVenta_(ctx) {
  const d = ctx.datos;
  const condicion = d.condicion === 'Contado' || d.condicion === 'Crédito' ? d.condicion : '';
  if (!condicion) throw errorUsuario_('Indique si la venta es al contado o a crédito.');
  if (typeof registrarVentaComoTicket_ === 'function') return registrarVentaComoTicket_(ctx);
  if (condicion === 'Crédito' && typeof registrarVentaCredito_ === 'function') return registrarVentaCredito_(ctx);

  const prep = prepararVenta_(ctx);
  const r = escribirVenta_(ctx, prep, { condicion: condicion });
  return {
    mensaje: 'Venta ' + r.id + ' registrada: ' + prep.cliente.nombre + ', Gs. ' + miles_(r.total) + ' (' + condicion.toLowerCase() + ').',
    id: r.id,
  };
}

function cambiarEstadoVenta_(ctx) {
  const d = ctx.datos;
  const tz = ctx.tz;
  const id = clave_(d.id);
  const ventas = leerTabla_(ctx.ss, 'ventas', tz);
  const filas = ventas.filas.filter(function (v) { return clave_(v.id) === id; });
  if (!id || !filas.length) throw errorUsuario_('No se encontró la venta ' + id + '.');
  const estado = filas[0].estado;
  const total = filas.reduce(function (t, f) { return t + f.subtotal; }, 0);
  const hoy = hoy_(tz);

  if (d.accion === 'cobrar') {
    if (estado === 'Anulada') throw errorUsuario_('La venta ' + id + ' está anulada; no se puede cobrar.');
    if (estado === 'Cobrada') return 'La venta ' + id + ' ya figuraba como cobrada.';
    // R0.5: una venta a crédito solo queda «Cobrada» cuando sus cobros registrados cancelan el saldo.
    if (filas[0].condicion === 'Crédito' && typeof registrarCobro_ === 'function') {
      throw errorUsuario_('Las ventas a crédito se cobran registrando el pago en «Cartera › Cobrar» (queda el recibo, ' +
        'se aplica a las cuotas y la venta pasa a «Cobrada» sola al cancelar el saldo).');
    }
    filas.forEach(function (f) { actualizarCeldas_(ventas, f._fila, { estado: 'Cobrada', fechaCobro: hoy }, tz); });
    return 'Venta ' + id + ' marcada como cobrada (Gs. ' + miles_(total) + ').';
  }

  if (d.accion === 'anular') {
    exigirAdministradora_(ctx, 'anular una venta');
    if (estado === 'Anulada') return 'La venta ' + id + ' ya estaba anulada.';
    const motivo = texto_(d.motivo, 200);
    if (motivo.length < 3) throw errorUsuario_('Escriba el motivo de la anulación.');
    if (filas[0].ticket && typeof anularTicket_ === 'function') {
      return anularTicket_(Object.assign({}, ctx, { datos: { id: filas[0].ticket, motivo: motivo } }));
    }
    return anularVentaFilas_(ctx, id, filas, motivo);
  }
  throw errorUsuario_('Acción no reconocida.');
}

/** Anula las filas de una venta: repone stock (salvo líneas de sistema como REC-FIN), anula su crédito y su comprobante. */
function anularVentaFilas_(ctx, id, filas, motivo) {
  const tz = ctx.tz;
  const ventas = leerTabla_(ctx.ss, 'ventas', tz);
  const hoy = hoy_(tz);
  const credito = filas[0].condicion === 'Crédito' && typeof anularCreditoDeVenta_ === 'function' ? anularCreditoDeVenta_(ctx, id, motivo) : '';
  const P = leerParametros_(ctx.ss, tz);
  const codRecargo = clave_(P.P_COD_REC || '');
  const productos = leerTabla_(ctx.ss, 'productos', tz);
  const movs = leerTabla_(ctx.ss, 'movimientos', tz);
  const stock = calcularStock_(movs.filas);
  let siguiente = siguienteNumero_(movs.filas.map(function (m) { return m.id; }), 'M-');
  const ahora = new Date();
  const reposiciones = filas.filter(function (f) {
    return (!codRecargo || clave_(f.codigo) !== codRecargo) && !!buscarPorCodigo_(productos.filas, f.codigo);
  }).map(function (f) {
    const k = clave_(f.codigo);
    stock[k] = (stock[k] || 0) + f.cantidad;
    return {
      id: formatoCodigo_('M-', siguiente++, 6), fecha: hoy, tipo: 'Anulación de venta', codigo: f.codigo,
      producto: f.producto, cantidad: f.cantidad, costoUnitario: f.costoUnitario, referencia: id,
      nota: motivo, usuario: ctx.email, registrado: ahora, unidad: f.unidad || 'DISTRIBUIDORA', ticket: f.ticket || '',
    };
  });
  actualizarFilas_(ventas, filas.map(function (f) {
    return { fila: f._fila, valores: { estado: 'Anulada', nota: (f.nota ? f.nota + ' · ' : '') + 'Anulada: ' + motivo } };
  }), tz);
  agregarFilas_(movs, reposiciones, tz);
  escribirStock_(productos, stock);
  const fiscal = typeof marcarComprobanteAnulado_ === 'function' ? marcarComprobanteAnulado_(ctx.ss, tz, id, motivo) : '';
  if (credito && typeof recalcularCartera_ === 'function') recalcularCartera_(ctx.ss, tz);
  return 'Venta ' + id + ' anulada. Se repuso el stock de sus productos.' + (credito ? ' ' + credito : '') + (fiscal ? ' ' + fiscal : '');
}

function guardarProfesional_(ctx) {
  const d = ctx.datos;
  const tabla = leerTabla_(ctx.ss, 'personal', ctx.tz);
  const original = clave_(d.codigoOriginal);
  const anterior = original ? buscarPorCodigo_(tabla.filas, original) : null;
  const comision = requerirPorcentaje_(d.comision, '% Comisión');
  const r = guardarCatalogo_(tabla, d.codigoOriginal, {
    nombre: requerirTexto_(d.nombre, 'Nombre', 80),
    rol: texto_(d.rol, 60),
    comision: comision,
    telefono: texto_(d.telefono, 40),
    activo: activoDe_(d.activo),
    actualizado: new Date(),
    usuario: ctx.email,
  }, 'E-', 3, 'El profesional', ctx.tz);
  // R3.6: todo cambio de % queda en «Historial de porcentajes» (con el módulo de comisiones).
  if (typeof anotarCambioPorcentaje_ === 'function' && (!anterior || Number(anterior.comision) !== comision)) {
    anotarCambioPorcentaje_(ctx, { codigoProfesional: r.codigo, profesional: d.nombre, porcentaje: comision, aplicacion: d.aplicacion, desde: d.desde });
  }

}

function guardarServicio_(ctx) {
  const d = ctx.datos;
  const tabla = leerTabla_(ctx.ss, 'servicios', ctx.tz);
  const vacio = d.comision === '' || d.comision === null || d.comision === undefined;
  const datos = {
    nombre: requerirTexto_(d.nombre, 'Servicio', 100),
    precio: requerirEntero_(d.precio, 'Precio base', 0, LIMITES.PRECIO_SERVICIO),
    comision: vacio ? 0 : requerirPorcentaje_(d.comision, '% Comisión del servicio'),
    activo: activoDe_(d.activo),
    actualizado: new Date(),
    usuario: ctx.email,
  };
  if (Object.prototype.hasOwnProperty.call(d, 'iva')) datos.iva = String(d.iva) === '' ? '' : textoTasa_(tasaIva_(d.iva, null));
  if (Object.prototype.hasOwnProperty.call(d, 'duracion')) datos.duracion = enteroOpcional_(d.duracion, 'Duración (min)', 720, true);
  if (!clave_(d.codigoOriginal)) {
    if (datos.iva === undefined) datos.iva = '10';
    datos.linea = 'SERVICIOS Y CAPACITACIÓN';
  }
  return guardarCatalogo_(tabla, d.codigoOriginal, datos, 'SV-', 3, 'El servicio', ctx.tz);
}

/**
 * Registrar un servicio (compatibilidad). Con el módulo de tickets (R4.5) crea el ticket internamente.
 * El % lo decide la regla vigente; solo una administradora puede indicar otro.
 */
function registrarServicio_(ctx) {
  if (typeof registrarServicioComoTicket_ === 'function') return registrarServicioComoTicket_(ctx);
  const d = ctx.datos;
  const tz = ctx.tz;
  const fecha = validarFecha_(d.fecha, tz, 'el servicio');
  const personal = leerTabla_(ctx.ss, 'personal', tz);
  const prof = buscarPorCodigo_(personal.filas, d.codigoProfesional);
  if (!prof) throw errorUsuario_('Seleccione el profesional que realizó el servicio.');
  if (!esActivo_(prof.activo)) throw errorUsuario_(prof.nombre + ' está inactivo en «Personal».');
  const servicios = leerTabla_(ctx.ss, 'servicios', tz);
  const serv = buscarPorCodigo_(servicios.filas, d.codigoServicio);
  if (!serv) throw errorUsuario_('Seleccione el servicio realizado.');
  if (!esActivo_(serv.activo)) throw errorUsuario_('El servicio «' + serv.nombre + '» está inactivo.');
  const precio = requerirEntero_(d.precio, 'Precio cobrado', 0, LIMITES.PRECIO_SERVICIO);
  const vacio = d.porcentaje === '' || d.porcentaje === null || d.porcentaje === undefined;
  if (!vacio) exigirAdministradora_(ctx, 'indicar un % de comisión distinto de la regla');
  const porcentaje = vacio ? porcentajeSugerido_(serv, prof) : requerirPorcentaje_(d.porcentaje, '% Comisión');
  const comision = gsEnteros_(precio * porcentaje / 100);
  const P = leerParametros_(ctx.ss, tz);
  const tasa = tasaIva_(serv.iva, P);

  const realizados = leerTabla_(ctx.ss, 'realizados', tz);
  const id = siguienteCodigo_(realizados.filas.map(function (r) { return r.id; }), 'S-', 5);
  agregarFilas_(realizados, [{
    id: id, fecha: fecha, codigoProfesional: prof.codigo, profesional: prof.nombre,
    codigoServicio: serv.codigo, servicio: serv.nombre, cliente: texto_(d.cliente, 80),
    precio: precio, porcentaje: porcentaje, comision: comision, estado: 'Realizado', liquidado: '',
    nota: texto_(d.nota, 300), usuario: ctx.email, registrado: new Date(),
    codigoCliente: clave_(d.codigoCliente), unidad: 'SALÓN', origen: 'NUEVO', precioLista: serv.precio || '',
    descuento: Math.round(descuentoLinea_(precio, Number(serv.precio) || 0) * 10000) / 100, iva: textoTasa_(tasa), ivaMonto: ivaIncluido_(precio, tasa),
  }], tz);
  return {
    mensaje: 'Servicio ' + id + ' registrado.' + (esAdministradora_(ctx.usuario) ? ' Comisión de ' + prof.nombre + ': Gs. ' + miles_(comision) +
      ' (' + numeroTexto_(porcentaje) + '%).' : ''),
    id: id,
  };
}

/** Regla vigente: si el servicio tiene % propio (> 0) se usa ese; si no, el % del profesional. */
function porcentajeSugerido_(serv, prof) {
  return serv.comision > 0 ? serv.comision : (prof.comision || 0);
}

function anularServicio_(ctx) {
  const d = ctx.datos;
  const tz = ctx.tz;
  const id = clave_(d.id);
  const realizados = leerTabla_(ctx.ss, 'realizados', tz);
  const fila = realizados.filas.find(function (r) { return clave_(r.id) === id; });
  if (!id || !fila) throw errorUsuario_('No se encontró el servicio ' + id + '.');
  if (fila.estado === 'Anulado') return 'El servicio ' + id + ' ya estaba anulado.';
  if (fila.liquidado) {
    throw errorUsuario_('La comisión del servicio ' + id + ' ya se liquidó el ' + formatoFecha_(fila.liquidado) + '; no se puede anular.');
  }
  const motivo = texto_(d.motivo, 200);
  if (motivo.length < 3) throw errorUsuario_('Escriba el motivo de la anulación.');
  if (fila.ticket && typeof anularLineaServicioDeTicket_ === 'function') return anularLineaServicioDeTicket_(ctx, fila, motivo);
  actualizarCeldas_(realizados, fila._fila, {
    estado: 'Anulado',
    nota: (fila.nota ? fila.nota + ' · ' : '') + 'Anulado: ' + motivo,
  }, tz);
  return 'Servicio ' + id + ' anulado. Su comisión ya no se contabiliza.';
}

/**
 * Liquida las comisiones pendientes del mes de una profesional: servicios, comisiones por venta de productos
 * en el salón (R4.7) y ajustes de recálculos sobre lo ya liquidado (R3.4).
 */
function liquidarComisiones_(ctx) {
  const d = ctx.datos;
  const tz = ctx.tz;
  const k = clave_(d.codigoProfesional);
  if (!k) throw errorUsuario_('Seleccione el profesional.');
  const mes = ctx.mes;
  const hoy = hoy_(tz);
  const realizados = leerTabla_(ctx.ss, 'realizados', tz);
  const pendientes = realizados.filas.filter(function (r) {
    return clave_(r.codigoProfesional) === k && r.fecha.slice(0, 7) === mes && r.estado !== 'Anulado' && !r.liquidado;
  });
  const personal = leerTabla_(ctx.ss, 'personal', tz);
  const prof = buscarPorCodigo_(personal.filas, k);
  const nombre = prof ? prof.nombre : (pendientes.length ? pendientes[0].profesional : k);
  let total = 0;
  let partes = 0;
  pendientes.forEach(function (r) { total += r.comision; });
  actualizarFilas_(realizados, pendientes.map(function (r) { return { fila: r._fila, valores: { liquidado: hoy } }; }), tz);
  partes += pendientes.length;
  // Comisión por venta de productos en el salón (líneas de «Ventas» con profesional).
  const ventas = leerTabla_(ctx.ss, 'ventas', tz);
  const deVentas = ventas.filas.filter(function (v) {
    return clave_(v.codigoProfesional) === k && v.fecha.slice(0, 7) === mes && v.estado !== 'Anulada' && !v.liquidado && v.comision > 0;
  });
  deVentas.forEach(function (v) { total += v.comision; });
  actualizarFilas_(ventas, deVentas.map(function (v) { return { fila: v._fila, valores: { liquidado: hoy } }; }), tz);
  partes += deVentas.length;
  // Ajustes de recálculos (positivos o negativos) todavía no liquidados.
  const ajustes = leerTabla_(ctx.ss, 'ajustes', tz);
  const deAjustes = ajustes.filas.filter(function (a) { return clave_(a.codigoProfesional) === k && !a.liquidado && !esActivo_(a.anulado); });
  deAjustes.forEach(function (a) { total += a.monto; });
  actualizarFilas_(ajustes, deAjustes.map(function (a) { return { fila: a._fila, valores: { liquidado: hoy } }; }), tz);
  partes += deAjustes.length;
  if (!partes) return 'No hay comisiones pendientes de ' + nombre + ' en ' + etiquetaMes_(mes) + '.';
  return 'Comisiones de ' + nombre + ' (' + etiquetaMes_(mes) + ') liquidadas: Gs. ' + miles_(total) + ' (' +
    pendientes.length + ' servicios' + (deVentas.length ? ', ' + deVentas.length + ' ventas de productos' : '') +
    (deAjustes.length ? ', ' + deAjustes.length + ' ajustes' : '') + ').';
}

/** Alta o edición genérica de catálogos con código automático y control de duplicados por nombre. */
function guardarCatalogo_(tabla, codigoOriginal, datos, prefijo, ancho, etiqueta, tz) {
  const original = clave_(codigoOriginal);
  let existente = null;
  if (original) {
    existente = buscarPorCodigo_(tabla.filas, original);
    if (!existente) throw errorUsuario_(etiqueta + ' ' + original + ' no existe. Actualice los datos e intente nuevamente.');
  }
  const duplicado = tabla.filas.find(function (f) {
    return f !== existente && normalizar_(f.nombre) === normalizar_(datos.nombre);
  });
  if (duplicado) throw errorUsuario_(etiqueta + ' «' + duplicado.nombre + '» ya existe (código ' + duplicado.codigo + ').');
  if (existente) {
    actualizarCeldas_(tabla, existente._fila, datos, tz);
    return { mensaje: 'Datos de «' + datos.nombre + '» actualizados.', codigo: existente.codigo };
  }
  datos.codigo = siguienteCodigo_(tabla.filas.map(function (f) { return f.codigo; }), prefijo, ancho);
  agregarFilas_(tabla, [datos], tz);
  return { mensaje: 'Se registró «' + datos.nombre + '» con el código ' + datos.codigo + '.', codigo: datos.codigo };
}

// ─────────────────────── 11. Datos para la aplicación ───────────────────────

/**
 * Arma los datos que recibe la aplicación según el rol (RT.2). A una OPERATIVA se le envía una LISTA BLANCA:
 * nunca costos, márgenes, valor del inventario, utilidades, comisiones de otras personas, condiciones de proveedores
 * ni el enlace a la hoja. Cada módulo de fase posterior aporta su parte solo si está instalado.
 */
function construirDatos_(ss, mes, tz, usuario) {
  const admin = esAdministradora_(usuario);
  const P = leerParametros_(ss, tz);
  const prod = leerTabla_(ss, 'productos', tz).filas;
  const movs = leerTabla_(ss, 'movimientos', tz).filas;
  const pelu = leerTabla_(ss, 'peluquerias', tz).filas;
  const vent = leerTabla_(ss, 'ventas', tz).filas;
  const stock = calcularStock_(movs);
  const tasaGeneral = P.P_IVA !== null ? Math.round(P.P_IVA * 1000) / 10 : 10;

  const productos = prod.map(function (p) {
    let tasa = tasaGeneral;
    try { tasa = p.iva === '' ? tasaGeneral : tasaIva_(p.iva, null); } catch (e) { tasa = tasaGeneral; }
    const o = {
      codigo: p.codigo, nombre: p.nombre, marca: p.marca, categoria: p.categoria, presentacion: p.presentacion,
      precio: p.precio, stockMinimo: p.stockMinimo, stock: stock[clave_(p.codigo)] || 0,
      activo: esActivo_(p.activo), iva: tasa, linea: p.linea,
    };
    if (admin) { o.costo = p.costo; o.proveedor = p.proveedor; }
    return o;
  }).sort(ordenarPor_(['marca', 'nombre']));

  const movimientos = movs.slice(-200).reverse().map(function (m) {
    return {
      id: m.id, fecha: m.fecha, tipo: m.tipo, codigo: m.codigo, producto: m.producto, cantidad: m.cantidad,
      referencia: m.referencia, nota: m.nota, usuario: m.usuario, unidad: m.unidad,
    };
  });

  const grupos = agruparVentas_(vent);
  const ventasMes = grupos.filter(function (v) { return v.fecha.slice(0, 7) === mes; }).sort(porFechaDesc_);
  const pendientes = grupos.filter(function (v) { return v.estado === 'Pendiente'; }).sort(porFechaAsc_);
  const ventasVigentes = ventasMes.filter(function (v) { return v.estado !== 'Anulada'; });
  const activos = productos.filter(function (p) { return p.activo; });

  const pers = leerTabla_(ss, 'personal', tz).filas;
  const serv = leerTabla_(ss, 'servicios', tz).filas;
  const real = leerTabla_(ss, 'realizados', tz).filas;
  const realizadosMes = real.filter(function (r) { return r.fecha.slice(0, 7) === mes; }).map(function (r) {
    const o = {
      id: r.id, fecha: r.fecha, codigoProfesional: r.codigoProfesional, profesional: r.profesional,
      codigoServicio: r.codigoServicio, servicio: r.servicio, cliente: r.cliente, codigoCliente: r.codigoCliente,
      precio: r.precio, estado: r.estado, nota: r.nota, ticket: r.ticket,
    };
    if (admin) { o.porcentaje = r.porcentaje; o.comision = r.comision; o.liquidado = r.liquidado; }
    return o;
  }).sort(porFechaDesc_);
  const serviciosVigentes = realizadosMes.filter(function (r) { return r.estado !== 'Anulado'; });

  const datos = {
    mes: mes,
    etiquetaMes: etiquetaMes_(mes),
    hoy: hoy_(tz),
    usuario: { email: usuario ? usuario.email : '', rol: usuario ? usuario.rol : ROLES.OPERATIVO, nombre: usuario ? usuario.nombre : '' },
    productos: productos,
    movimientos: movimientos,
    peluquerias: pelu.map(function (c) {
      return {
        codigo: c.codigo, nombre: c.nombre, contacto: c.contacto, telefono: c.telefono, ciudad: c.ciudad,
        direccion: c.direccion, ruc: c.ruc, activo: esActivo_(c.activo), ci: c.ci, observacion: c.observacion,
        razonSocial: c.razonSocial, celular: c.celular, correo: c.correo, tipo: c.tipo, categoria: c.categoria,
        zona: c.zona, legado: c.legado, alias: c.alias, completitud: c.completitud, faltantes: c.faltantes,
        limite: c.limite, saldo: c.saldo, vencido: c.vencido, atraso: c.atraso, estadoCredito: c.estadoCredito,
        disponible: c.disponible, sugerencia: admin ? c.sugerencia : '',
      };
    }).sort(ordenarPor_(['nombre'])),
    ventas: ventasMes,
    pendientes: pendientes,
    marcas: unicos_(CONFIG.MARCAS_BASE.concat(productos.map(function (p) { return p.marca; }))),
    categorias: unicos_(productos.map(function (p) { return p.categoria; })),
    tiposMovimiento: Object.keys(TIPOS_MOVIMIENTO).map(function (k) {
      return { clave: k, etiqueta: TIPOS_MOVIMIENTO[k].etiqueta, signo: TIPOS_MOVIMIENTO[k].signo, notaObligatoria: TIPOS_MOVIMIENTO[k].notaObligatoria };
    }),
    personal: pers.map(function (p) {
      const o = { codigo: p.codigo, nombre: p.nombre, rol: p.rol, telefono: p.telefono, activo: esActivo_(p.activo) };
      if (admin) o.comision = p.comision;
      return o;
    }).sort(ordenarPor_(['nombre'])),
    servicios: serv.map(function (s) {
      const o = { codigo: s.codigo, nombre: s.nombre, precio: s.precio, activo: esActivo_(s.activo), duracion: s.duracion || '', iva: s.iva, linea: s.linea };
      if (admin) o.comision = s.comision;
      return o;
    }).sort(ordenarPor_(['nombre'])),
    realizados: realizadosMes,
    comisiones: [],
    listas: { medios: LISTAS.MEDIOS_PAGO, unidades: LISTAS.UNIDADES, tiposCliente: LISTAS.TIPOS_CLIENTE, zonas: LISTAS.ZONAS,
      lineas: LISTAS.LINEAS, comprobantes: LISTAS.COMPROBANTES, frecuencias: LISTAS.FRECUENCIAS, conceptos: LISTAS.CONCEPTOS_COBRO },
    config: {
      registroAbierto: registroAbierto_(ss), version: CONFIG.VERSION,
      sifen: modoIntegracion_(ss, 'sifen'), whatsapp: modoIntegracion_(ss, 'whatsapp'), facturacion: modoFacturacion_(ss),
      modulos: modulosInstalados_(),
    },
    kpis: {
      stockBajo: activos.filter(function (p) { return p.stock <= p.stockMinimo; }).length,
      productosActivos: activos.length,
      cantidadVentas: ventasVigentes.length,
      cantidadServicios: serviciosVigentes.length,
      porCobrar: suma_(pendientes, 'total'),
      cantidadPorCobrar: pendientes.length,
    },
  };

  // Módulos de las fases siguientes (cada uno aplica su propia lista blanca según el rol).
  const contexto = { P: P, productos: prod, ventas: vent, grupos: grupos, peluquerias: pelu, realizados: real, personal: pers, servicios: serv, mes: mes };
  if (typeof datosCartera_ === 'function') datos.cartera = datosCartera_(ss, tz, usuario, contexto);
  if (typeof datosClientes_ === 'function') datos.clientes = datosClientes_(ss, tz, usuario, contexto);
  if (typeof datosTickets_ === 'function') datos.tickets = datosTickets_(ss, tz, usuario, contexto);
  if (typeof datosAgenda_ === 'function') datos.agenda = datosAgenda_(ss, tz, usuario, contexto);
  if (datos.cartera && datos.cartera.totales) {
    datos.kpis.porCobrar = datos.cartera.totales.cartera;
    datos.kpis.cantidadPorCobrar = datos.cartera.totales.creditosConSaldo;
    datos.kpis.carteraVencida = datos.cartera.totales.vencido;
  }

  if (!admin) return datos;

  // ── Solo administradoras ──
  datos.comisiones = resumenComisiones_(realizadosMes, pers);
  datos.kpis.ventasMes = suma_(ventasVigentes, 'total');
  datos.kpis.serviciosMes = suma_(serviciosVigentes, 'precio');
  datos.kpis.comisionesMes = suma_(serviciosVigentes, 'comision');
  datos.kpis.comisionesPendientes = suma_(serviciosVigentes.filter(function (r) { return !r.liquidado; }), 'comision');
  datos.kpis.valorInventario = activos.reduce(function (t, p) { return t + Math.max(p.stock, 0) * (p.costo || 0); }, 0);
  datos.config.admin = CONFIG.CORREO_ADMIN;
  datos.pendientesPanel = pendientesSistema_(ss, tz, contexto);
  if (typeof datosProveedores_ === 'function') datos.proveedores = datosProveedores_(ss, tz, usuario, contexto);
  if (typeof datosComisiones_ === 'function') datos.recalculo = datosComisiones_(ss, tz, usuario, contexto);
  if (typeof datosAdmin_ === 'function') datos.admin = datosAdmin_(ss, tz, usuario, contexto);
  return datos;
}

/**
 * Panel «Pendientes» (solo administradoras): lo que falta completar o confirmar, con su cantidad. Nunca se completa
 * con supuestos. Cada módulo agrega los suyos (clientes incompletos, saldos dudosos, acuerdos por vencer…).
 */
function pendientesSistema_(ss, tz, c) {
  const lista = [];
  function agregar(clave, titulo, n, detalle, accion) {
    if (n > 0) lista.push({ clave: clave, titulo: titulo, cantidad: n, detalle: (detalle || []).slice(0, 30), accion: accion || '' });
  }
  const activos = c.productos.filter(function (p) { return esActivo_(p.activo); });
  const sinCosto = activos.filter(function (p) { return !(p.costo > 0); });
  agregar('costos', 'Productos sin costo cargado', sinCosto.length, sinCosto.map(function (p) { return p.codigo + ' ' + p.nombre; }), 'Editar el producto (Inventario).');
  const sinLinea = activos.filter(function (p) { return !p.linea || p.linea === 'REVISAR'; });
  agregar('lineas', 'Productos sin línea / centro de costo (o en REVISAR)', sinLinea.length, sinLinea.map(function (p) { return p.codigo + ' ' + p.nombre + ' (' + (p.categoria || 'sin categoría') + ')'; }), 'Asignar la línea en Inventario.');
  const sinIva = activos.filter(function (p) { return String(p.iva) === ''; });
  agregar('iva', 'Productos sin tasa de IVA confirmada', sinIva.length, sinIva.map(function (p) { return p.codigo + ' ' + p.nombre; }), 'Confirmar 10%, 5% o exenta con el contador.');
  agregar('marcas', 'Marcas escritas de distintas formas', duplicadosPorMayusculas_(c.productos.map(function (p) { return p.marca; })).length,
    duplicadosPorMayusculas_(c.productos.map(function (p) { return p.marca; })), 'Unificar el nombre en Inventario.');
  agregar('categorias', 'Categorías de producto duplicadas o sin normalizar', duplicadosPorMayusculas_(c.productos.map(function (p) { return p.categoria; })).length,
    duplicadosPorMayusculas_(c.productos.map(function (p) { return p.categoria; })), 'Unificar el nombre en Inventario.');
  const sinDuracion = c.servicios.filter(function (s) { return esActivo_(s.activo) && !(s.duracion > 0); });
  agregar('duracion', 'Servicios sin duración', sinDuracion.length, sinDuracion.map(function (s) { return s.codigo + ' ' + s.nombre; }), 'Cargar la duración (min) en Salón › Servicios.');
  agregar('parametros', 'Parámetros de crédito vacíos o inválidos', c.P._faltan.length, c.P._faltan.map(function (k) { return c.P._nombres[k] || k; }), 'Completar en «Parámetros de crédito».');
  const opcionales = ['P_META_ASISTENTE', 'P_REPOSICION', 'P_POLITICA_FIRMADA', 'P_IVA_RECARGO'].filter(function (k) { return c.P[k] === null; });
  agregar('pendientesMirian', 'Datos por confirmar con Mirian o el contador', opcionales.length, opcionales.map(function (k) { return c.P._nombres[k] || k; }), 'Ver «Parámetros de crédito».');
  let entidades = [];
  try { entidades = leerTabla_(ss, 'entidades', tz).filas.filter(function (e) { return e.estado !== 'COMPLETO'; }); } catch (e) { entidades = []; }
  agregar('fiscal', 'Datos fiscales pendientes (Entidades fiscales)', entidades.length, entidades.map(function (e) { return e.unidad; }), 'Completar con el contador.');
  if (typeof pendientesCartera_ === 'function') pendientesCartera_(ss, tz, c).forEach(function (p) { lista.push(p); });
  if (typeof pendientesClientes_ === 'function') pendientesClientes_(ss, tz, c).forEach(function (p) { lista.push(p); });
  if (typeof pendientesProveedores_ === 'function') pendientesProveedores_(ss, tz, c).forEach(function (p) { lista.push(p); });
  if (typeof pendientesComisiones_ === 'function') pendientesComisiones_(ss, tz, c).forEach(function (p) { lista.push(p); });
  return lista;
}

/** Valores que difieren solo en mayúsculas, tildes o plural simple («Herramientas» / «Herramienta»). */
function duplicadosPorMayusculas_(valores) {
  const grupos = {};
  valores.forEach(function (v) {
    const s = String(v || '').trim();
    if (!s) return;
    const k = normalizar_(s).replace(/s$/, '');
    (grupos[k] = grupos[k] || {})[s] = true;
  });
  return Object.keys(grupos).filter(function (k) { return Object.keys(grupos[k]).length > 1; }).map(function (k) { return Object.keys(grupos[k]).join(' / '); });
}

function agruparVentas_(filas) {
  const mapa = {};
  const lista = [];
  filas.forEach(function (f) {
    const id = clave_(f.id);
    if (!id) return;
    let v = mapa[id];
    if (!v) {
      v = mapa[id] = {
        id: f.id, fecha: f.fecha, codigoCliente: f.codigoCliente, cliente: f.cliente, condicion: f.condicion,
        estado: f.estado, fechaCobro: f.fechaCobro, nota: f.nota, usuario: f.usuario, total: 0, unidades: 0, items: [],
        ticket: f.ticket || '', unidad: f.unidad || '', origen: f.origen || '', nroCredito: f.nroCredito || '', autorizadoPor: f.autorizadoPor || '',
      };
      lista.push(v);
    }
    v.items.push({ codigo: f.codigo, producto: f.producto, cantidad: f.cantidad, precio: f.precio, subtotal: f.subtotal, iva: f.iva, precioLista: f.precioLista });
    v.total += f.subtotal;
    v.unidades += f.cantidad;
  });
  return lista;
}

function resumenComisiones_(realizadosMes, personal) {
  const mapa = {};
  const orden = [];
  realizadosMes.forEach(function (r) {
    if (r.estado === 'Anulado') return;
    const k = clave_(r.codigoProfesional);
    if (!mapa[k]) {
      const p = buscarPorCodigo_(personal, k);
      mapa[k] = {
        codigo: r.codigoProfesional, nombre: p ? p.nombre : r.profesional, rol: p ? p.rol : '',
        servicios: 0, facturado: 0, comision: 0, liquidado: 0, pendiente: 0,
      };
      orden.push(mapa[k]);
    }
    const m = mapa[k];
    m.servicios += 1;
    m.facturado += r.precio;
    m.comision += r.comision;
    if (r.liquidado) m.liquidado += r.comision; else m.pendiente += r.comision;
  });
  return orden.sort(function (a, b) { return b.comision - a.comision; });
}

// ─────────────────────── 12. Utilidades ───────────────────────

function normalizarCorreo_(valor, esSesion) {
  const correo = String(valor === null || valor === undefined ? '' : valor).trim().toLowerCase();
  if (!correo) throw errorUsuario_((esSesion ? PREFIJO_SESION : '') + 'Ingrese su correo electrónico.');
  if (correo.length > 254 || !/^[^\s@,;:<>()\[\]"']+@[^\s@,;:<>()\[\]"']+\.[a-z]{2,}$/i.test(correo)) {
    throw errorUsuario_((esSesion ? PREFIJO_SESION : '') + 'El correo «' + String(valor).trim() + '» no es válido. Revíselo e intente nuevamente.');
  }
  return correo;
}

function normalizarMes_(valor) {
  const s = String(valor || '');
  if (/^\d{4}-(0[1-9]|1[0-2])$/.test(s)) return s;
  return Utilities.formatDate(new Date(), CONFIG.ZONA_HORARIA, 'yyyy-MM');
}

function clave_(v) {
  return String(v === null || v === undefined ? '' : v).trim().toUpperCase();
}

function normalizar_(s) {
  return String(s || '').normalize('NFD').replace(/[̀-ͯ]/g, '').toLowerCase().replace(/\s+/g, ' ').trim();
}

function buscarPorCodigo_(filas, codigo) {
  const k = clave_(codigo);
  if (!k) return null;
  return filas.find(function (f) { return clave_(f.codigo) === k; }) || null;
}

function esActivo_(v) {
  return !/^(no|n|false|falso|0|inactivo)$/i.test(String(v === null || v === undefined ? '' : v).trim());
}

function activoDe_(v) {
  return v === false || /^(no|false|0)$/i.test(String(v === null || v === undefined ? '' : v).trim()) ? 'No' : 'Sí';
}

function unicos_(lista) {
  const vistos = {};
  const salida = [];
  lista.forEach(function (v) {
    const s = String(v === null || v === undefined ? '' : v).trim();
    const k = normalizar_(s);
    if (s && !vistos[k]) {
      vistos[k] = true;
      salida.push(s);
    }
  });
  return salida;
}

function ordenarPor_(campos) {
  return function (a, b) {
    for (let i = 0; i < campos.length; i++) {
      const x = normalizar_(a[campos[i]]);
      const y = normalizar_(b[campos[i]]);
      if (x < y) return -1;
      if (x > y) return 1;
    }
    return 0;
  };
}

function porFechaDesc_(a, b) {
  if (a.fecha !== b.fecha) return a.fecha < b.fecha ? 1 : -1;
  return String(a.id) < String(b.id) ? 1 : -1;
}

function porFechaAsc_(a, b) {
  return -porFechaDesc_(a, b);
}

function suma_(lista, campo) {
  return lista.reduce(function (t, x) { return t + (Number(x[campo]) || 0); }, 0);
}

function texto_(valor, max) {
  let s = String(valor === null || valor === undefined ? '' : valor)
    .replace(/[\u0000-\u001F\u007F]/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();
  if (max && s.length > max) s = s.slice(0, max).trim();
  return s;
}

function requerirTexto_(valor, etiqueta, max) {
  const s = texto_(valor, max);
  if (!s) throw errorUsuario_('Complete el campo «' + etiqueta + '».');
  return s;
}

/** Acepta números o textos como "85.000" (separador de miles paraguayo). */
function entero_(valor) {
  if (typeof valor === 'number') return isFinite(valor) ? Math.round(valor) : NaN;
  const s = String(valor === null || valor === undefined ? '' : valor).trim();
  if (!s) return NaN;
  const digitos = s.replace(/[^0-9]/g, '');
  if (!digitos) return NaN;
  const n = parseInt(digitos, 10);
  return /^-/.test(s) ? -n : n;
}

/** Acepta "37,5", "37.5" o "37,5 %". */
function decimal_(valor) {
  if (typeof valor === 'number') return isFinite(valor) ? valor : NaN;
  let s = String(valor === null || valor === undefined ? '' : valor).replace(/[%\s]/g, '');
  if (!s) return NaN;
  if (s.indexOf(',') >= 0 && s.indexOf('.') >= 0) s = s.replace(/\./g, '').replace(',', '.');
  else s = s.replace(',', '.');
  const n = Number(s);
  return isFinite(n) ? n : NaN;
}

function requerirEntero_(valor, etiqueta, min, max) {
  const n = entero_(valor);
  if (isNaN(n)) throw errorUsuario_('Ingrese un número válido en «' + etiqueta + '».');
  if (min !== null && min !== undefined && n < min) throw errorUsuario_('«' + etiqueta + '» no puede ser menor que ' + miles_(min) + '.');
  if (max !== null && max !== undefined && n > max) throw errorUsuario_('«' + etiqueta + '» no puede ser mayor que ' + miles_(max) + '. Revise si sobra algún cero.');
  return n;
}

/** Entero ≥ 0 opcional: vacío = 0 (o null si devolverNulo). */
function enteroOpcional_(valor, etiqueta, max, devolverNulo) {
  if (valor === '' || valor === null || valor === undefined) return devolverNulo ? null : 0;
  return requerirEntero_(valor, etiqueta, 0, max);
}

function requerirPorcentaje_(valor, etiqueta) {
  const n = decimal_(valor);
  if (isNaN(n) || n < 0 || n > 100) throw errorUsuario_('«' + etiqueta + '» debe ser un porcentaje entre 0 y 100.');
  return Math.round(n * 100) / 100;
}

function siguienteNumero_(lista, prefijo) {
  const re = new RegExp('^' + prefijo.replace(/[-\/\\^$*+?.()|[\]{}]/g, '\\$&') + '(\\d+)$', 'i');
  let max = 0;
  lista.forEach(function (v) {
    const m = String(v || '').trim().match(re);
    if (m) max = Math.max(max, parseInt(m[1], 10));
  });
  return max + 1;
}

function formatoCodigo_(prefijo, n, ancho) {
  return prefijo + String(n).padStart(ancho, '0');
}

function siguienteCodigo_(lista, prefijo, ancho) {
  return formatoCodigo_(prefijo, siguienteNumero_(lista, prefijo), ancho);
}

function miles_(n) {
  const v = Math.round(Number(n) || 0);
  return (v < 0 ? '-' : '') + String(Math.abs(v)).replace(/\B(?=(\d{3})+(?!\d))/g, '.');
}

function numeroTexto_(n) {
  return String(n).replace('.', ',');
}

function hoy_(tz) {
  return Utilities.formatDate(new Date(), tz, 'yyyy-MM-dd');
}

function etiquetaMes_(mes) {
  const p = String(mes).split('-');
  return MESES_[Number(p[1]) - 1] + ' de ' + p[0];
}

function formatoFecha_(iso) {
  const p = String(iso || '').split('-');
  return p.length === 3 ? p[2] + '/' + p[1] + '/' + p[0] : String(iso || '');
}

function sumarDias_(iso, dias) {
  const p = iso.split('-').map(Number);
  const f = new Date(Date.UTC(p[0], p[1] - 1, p[2] + dias));
  return f.getUTCFullYear() + '-' + ('0' + (f.getUTCMonth() + 1)).slice(-2) + '-' + ('0' + f.getUTCDate()).slice(-2);
}

/** Valida 'AAAA-MM-DD': fecha real, no futura y no anterior a 2015. */
function validarFecha_(valor, tz, etiqueta) {
  const s = String(valor || '').trim();
  if (!/^\d{4}-\d{2}-\d{2}$/.test(s)) throw errorUsuario_('Ingrese una fecha válida para ' + etiqueta + '.');
  const p = s.split('-').map(Number);
  const f = new Date(Date.UTC(p[0], p[1] - 1, p[2]));
  if (f.getUTCFullYear() !== p[0] || f.getUTCMonth() !== p[1] - 1 || f.getUTCDate() !== p[2]) {
    throw errorUsuario_('La fecha ' + formatoFecha_(s) + ' no existe.');
  }
  if (s > sumarDias_(hoy_(tz), 1)) throw errorUsuario_('No se permiten fechas futuras (' + formatoFecha_(s) + '). Revise la fecha.');
  if (s < '2015-01-01') throw errorUsuario_('Revise el año de la fecha (' + formatoFecha_(s) + ').');
  return s;
}

function fechaATexto_(valor, tz) {
  if (valor instanceof Date) return isNaN(valor.getTime()) ? '' : Utilities.formatDate(valor, tz, 'yyyy-MM-dd');
  const s = String(valor === null || valor === undefined ? '' : valor).trim();
  let m = s.match(/^(\d{4})-(\d{2})-(\d{2})/);
  if (m) return m[1] + '-' + m[2] + '-' + m[3];
  m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})/);
  if (m) return m[3] + '-' + ('0' + m[2]).slice(-2) + '-' + ('0' + m[1]).slice(-2);
  return '';
}

function fechaHoraATexto_(valor, tz) {
  if (valor instanceof Date) return isNaN(valor.getTime()) ? '' : Utilities.formatDate(valor, tz, 'yyyy-MM-dd HH:mm');
  return String(valor === null || valor === undefined ? '' : valor).trim();
}

/** 'AAAA-MM-DD' → fecha a las 12:00 en la zona de la hoja (evita corrimientos de día). */
function textoAFecha_(iso, tz) {
  return Utilities.parseDate(iso + ' 12:00', tz, 'yyyy-MM-dd HH:mm');
}

/** v2.2: 'AAAA-MM-DD HH:mm' (o solo la fecha) → fecha y hora en la zona de la hoja. */
function textoAFechaHora_(texto, tz) {
  const s = String(texto).trim();
  return Utilities.parseDate(s.length === 10 ? s + ' 00:00' : s, tz, 'yyyy-MM-dd HH:mm');
}

/** v2.2: días entre dos fechas 'AAAA-MM-DD' (b − a). */
function diasEntre_(a, b) {
  const p = String(a).split('-').map(Number);
  const q = String(b).split('-').map(Number);
  return Math.round((Date.UTC(q[0], q[1] - 1, q[2]) - Date.UTC(p[0], p[1] - 1, p[2])) / 86400000);
}

/**
 * Aplica los paquetes pendientes y devuelve los avisos para quien ingresa (cada aviso se muestra una sola vez por correo).
 * Los avisos de paquetes «soloAdministradoras» no se muestran a las operativas. Devuelve también el detalle en
 * aplicarPaquetes_.ultimo (para el informe de instalación).
 */
function aplicarPaquetes_(ss, email, tz, rol) {
  const props = PropertiesService.getScriptProperties();
  const aplicados = leerJsonPropiedad_(props, CLAVE_PAQUETES);
  const avisados = leerJsonPropiedad_(props, CLAVE_PAQUETES_AVISADOS);
  const avisos = [];
  const detalle = [];
  const idHoja = ss.getId();
  PAQUETES.forEach(function (paq) {
    const clave = paq.id + '|' + idHoja;
    if (!aplicados[clave]) {
      if (paq.disponible && !paq.disponible()) { detalle.push({ id: paq.id, estado: 'NO INSTALADO' }); return; }
      if (paq.detectarAplicado && paq.detectarAplicado(ss)) {

        return;
      }
      try {
        const resumen = paq.aplicar(ss, email, tz);
        SpreadsheetApp.flush();

        console.error('Paquete ' + paq.id + ': ' + (e && e.stack ? e.stack : e));
        detalle.push({ id: paq.id, estado: 'ERROR', resumen: String(e && e.message ? e.message : e) });
        avisos.push('No se pudo completar la actualización del ' + paq.titulo + ' (' + String(e && e.message ? e.message : e) + '). Se reintentará sola.');

      }
    } else {
      detalle.push({ id: paq.id, estado: 'YA APLICADO' });
    }
    const vistos = avisados[clave] || [];
    if (email && vistos.indexOf(email) < 0) {
      const esAdmin = rol === ROLES.ADMIN;
      if (paq.aviso) avisos.push(paq.aviso);
      else if (!paq.soloAdministradoras || esAdmin) avisos.push('Actualización aplicada: ' + aplicados[clave].resumen);
      vistos.push(email);
      avisados[clave] = vistos;
      props.setProperty(CLAVE_PAQUETES_AVISADOS, JSON.stringify(avisados));
    }
  });
  aplicarPaquetes_.ultimo = detalle;
  return avisos;
}

function leerJsonPropiedad_(props, clave) {
  try {
    const v = JSON.parse(props.getProperty(clave) || '{}');
    return v && typeof v === 'object' ? v : {};
  } catch (e) {
    return {};
  }
}

/** Paquete del salón: catálogo de servicios + hojas de referencia + control mensual. */
function aplicarPaqueteSalon_(ss, email, tz) {
  if (typeof DATOS_SALON === 'undefined') throw new Error('Falta el archivo DatosSalon.gs (datos del salón de la v2.1).');
  const D = DATOS_SALON;
  const r = cargarServiciosSalon_(ss, D.servicios, tz);
  D.hojas.forEach(function (spec) { escribirHojaReferencia_(ss, spec, D.estilos); });
  const tarifario = ss.getSheetByName(D.tarifario.hoja);
  if (tarifario && r.codigos.length) {
    tarifario.getRange(D.tarifario.fila, D.tarifario.columna, r.codigos.length, 1)
      .setValues(r.codigos.map(function (c) { return [c]; }));
  }
  crearControlMensualSalon_(ss, D.control);
  anotarLeameYVersion_(ss);
  const resumen = D.servicios.length + ' servicios del salón en «Servicios» (' + r.creados + ' nuevos, ' + r.actualizados +
    ' actualizados) y ' + (D.hojas.length + 2) + ' hojas «Salón · …» con el tarifario, las comisiones, la rentabilidad y el control mensual. «Personal» no se modificó.';
  const registro = r.cambios.map(function (c) { return [D.version, c[0], c[1], email]; });
  registro.push([D.version, 'Hojas de referencia', D.hojas.map(function (h) { return h.n; }).join(', ') + ', ' + HOJA_CONTROL_SALON, email]);
  registro.push([D.version, 'Paquete aplicado', resumen, email]);
  registrarActualizaciones_(ss, registro);
  return resumen;
}

/** Agrega o actualiza los servicios del tarifario en «Servicios» (por nombre, sin duplicar). */
function cargarServiciosSalon_(ss, lista, tz) {
  const tabla = leerTabla_(ss, 'servicios', tz);
  let siguiente = siguienteNumero_(tabla.filas.map(function (f) { return f.codigo; }), 'SV-');
  const ahora = new Date();
  const nuevos = [], codigos = [], cambios = [];
  let creados = 0, actualizados = 0;
  lista.forEach(function (s) {
    const precio = Math.round(Number(s.precio)) || 0;
    const comision = Math.round((Number(s.comision) || 0) * 100) / 100;
    const textoPct = comision > 0 ? numeroTexto_(comision) + '%' : '% de cada profesional';
    const existente = tabla.filas.find(function (f) { return normalizar_(f.nombre) === normalizar_(s.nombre); });
    if (existente) {
      codigos.push(existente.codigo);
      if (existente.precio !== precio || Math.abs((existente.comision || 0) - comision) > 0.001) {
        actualizarCeldas_(tabla, existente._fila, { precio: precio, comision: comision, actualizado: ahora, usuario: USUARIO_ACTUALIZACION }, tz);
        cambios.push(['Servicio actualizado', existente.codigo + ' · ' + existente.nombre + ': precio Gs. ' + miles_(existente.precio) + ' → ' +
          miles_(precio) + '; comisión ' + (existente.comision > 0 ? numeroTexto_(existente.comision) + '%' : '% de cada profesional') + ' → ' + textoPct]);
        actualizados++;
      }
      return;
    }
    const codigo = formatoCodigo_('SV-', siguiente++, 3);
    codigos.push(codigo);
    nuevos.push({ codigo: codigo, nombre: s.nombre, precio: precio, comision: comision, activo: 'Sí', actualizado: ahora, usuario: USUARIO_ACTUALIZACION });
    cambios.push(['Servicio creado', codigo + ' · ' + s.nombre + ': Gs. ' + miles_(precio) + ' · comisión: ' + textoPct]);
    creados++;
  });
  agregarFilas_(tabla, nuevos, tz);
  return { codigos: codigos, cambios: cambios, creados: creados, actualizados: actualizados };
}

/** Recrea una hoja de referencia a partir de su especificación (valores, formatos, combinaciones y anchos). */
function escribirHojaReferencia_(ss, spec, estilos) {
  let hoja = ss.getSheetByName(spec.n);
  const conservados = [];
  if (hoja) {
    (spec.k || []).forEach(function (a1) { conservados.push([a1, hoja.getRange(a1).getValues()]); });
    quitarProtecciones_(hoja);
    hoja.getRange(1, 1, hoja.getMaxRows(), hoja.getMaxColumns()).breakApart();
    hoja.clear();
    hoja.getRange(1, 1, hoja.getMaxRows(), hoja.getMaxColumns()).clearDataValidations();
  } else {
    hoja = ss.insertSheet(spec.n, ss.getSheets().length);
  }
  const nf = spec.v.length;
  const nc = spec.nc;
  asegurarTamano_(hoja, nf, nc);

  const formatos = [], valores = [], tamanos = [], negritas = [], cursivas = [], ajustes = [];
  const colores = {}, fondos = {}, horizontales = {}, verticales = {}, conBorde = [];
  for (let i = 0; i < nf; i++) {
    const fila = spec.v[i];
    const ids = [];
    (spec.s[i] || []).forEach(function (par) { for (let k = 0; k < par[1]; k++) ids.push(par[0]); });
    while (ids.length < nc) ids.push(0);
    const fFmt = [], fVal = [], fTam = [], fNeg = [], fCur = [], fAju = [];
    for (let j = 0; j < nc; j++) {
      const e = estilos[ids[j]];
      const v = j < fila.length ? fila[j] : '';
      fVal.push(v);
      fFmt.push(typeof v === 'number' ? e[0] : '@');
      fTam.push(e[3]);
      fNeg.push(e[1] ? 'bold' : 'normal');
      fCur.push(e[2] ? 'italic' : 'normal');
      fAju.push(!!e[6]);
    }
    formatos.push(fFmt); valores.push(fVal); tamanos.push(fTam); negritas.push(fNeg); cursivas.push(fCur); ajustes.push(fAju);
    agruparTramos_(colores, tramosFila_(ids, estilos, 4, i + 1, '#000000'));
    agruparTramos_(fondos, tramosFila_(ids, estilos, 5, i + 1, null));
    agruparTramos_(horizontales, tramosFila_(ids, estilos, 7, i + 1, null));
    agruparTramos_(verticales, tramosFila_(ids, estilos, 8, i + 1, null));
    tramosFila_(ids, estilos, 9, i + 1, false).forEach(function (t) { conBorde.push(t[1]); });
  }
  const rango = hoja.getRange(1, 1, nf, nc);
  intentar_('formatos de número', function () { rango.setNumberFormats(formatos); });
  rango.setValues(valores);
  intentar_('fuentes', function () {
    rango.setFontFamily('Arial').setFontSizes(tamanos).setFontWeights(negritas).setFontStyles(cursivas).setFontColor('#000000');
  });
  intentar_('ajuste de texto', function () { rango.setWraps(ajustes).setVerticalAlignment('middle'); });
  aplicarGrupos_(hoja, colores, 'colores', function (rl, v) { rl.setFontColor(v); });
  aplicarGrupos_(hoja, fondos, 'fondos', function (rl, v) { rl.setBackground(v); });
  aplicarGrupos_(hoja, horizontales, 'alineación', function (rl, v) { rl.setHorizontalAlignment(v); });
  aplicarGrupos_(hoja, verticales, 'alineación vertical', function (rl, v) { rl.setVerticalAlignment(v); });
  if (conBorde.length) {
    aplicarGrupos_(hoja, { borde: conBorde }, 'bordes', function (rl) {
      rl.setBorder(true, true, true, true, true, true, '#D9D4CA', SpreadsheetApp.BorderStyle.SOLID);
    });
  }
  const merges = (spec.m || []).map(rangoNumerico_);
  merges.forEach(function (m) {
    intentar_('combinar ' + m.a1, function () { hoja.getRange(m.r1, m.c1, m.r2 - m.r1 + 1, m.c2 - m.c1 + 1).merge(); });
  });
  (spec.w || []).forEach(function (px, j) {
    if (px > 0) intentar_('ancho', function () { hoja.setColumnWidth(j + 1, px); });
  });
  (spec.h || []).forEach(function (par) { intentar_('alto', function () { hoja.setRowHeight(par[0], par[1]); }); });
  const fr = merges.some(function (m) { return m.r1 <= spec.fr && spec.fr < m.r2; }) ? 0 : (spec.fr || 0);
  const fc = merges.some(function (m) { return m.c1 <= spec.fc && spec.fc < m.c2; }) ? 0 : (spec.fc || 0);
  intentar_('inmovilizar', function () { hoja.setFrozenRows(fr); hoja.setFrozenColumns(fc); });
  if (spec.g === false) intentar_('líneas de cuadrícula', function () { hoja.setHiddenGridlines(true); });
  intentar_('color de pestaña', function () { hoja.setTabColor(COLOR_SALON); });
  conservados.forEach(function (par) { hoja.getRange(par[0]).setValues(par[1]); });
  const descripcion = 'Salón: hoja de referencia (copia del archivo Excel)';
  if (spec.p === 'hoja') {
    hoja.protect().setDescription(descripcion).setWarningOnly(true);
  } else if (spec.p === 'rangos') {
    (spec.pr || []).forEach(function (a1) { hoja.getRange(a1).protect().setDescription(descripcion).setWarningOnly(true); });
  }
  return hoja;
}

/** Agrega líneas en «Léame» (sin repetir) y actualiza «Versión del sistema» en «Configuración». */
function anotarLeameYVersion_(ss, lineas) {
  const leame = ss.getSheetByName(HOJA_LEAME);
  if (leame) {
    const textos = lineas && lineas.length ? lineas
      : ['• Hojas «Salón · …»: tarifario, comisiones, costos, rentabilidad, mercado y marco legal del salón (copia de referencia); «Salón · Control mensual» se completa solo con los registros del sistema.'];
    textos.forEach(function (texto) {
      const ultima = Math.max(leame.getLastRow(), 1);
      const existentes = leame.getRange(1, 1, ultima, 1).getValues().map(function (f) { return String(f[0]); });
      if (existentes.indexOf(texto) < 0) leame.getRange(ultima + 1, 1).setValue(texto).setWrap(true);
    });
  }
  const conf = ss.getSheetByName(HOJA_CONFIG);
  if (conf && conf.getLastRow() >= 1) {
    const v = conf.getRange(1, 1, conf.getLastRow(), 1).getValues();
    for (let i = 0; i < v.length; i++) {
      if (String(v[i][0]).trim() === 'Versión del sistema') {
        conf.getRange(i + 1, 2).setValue(celdaTexto_(CONFIG.VERSION));
        break;
      }
    }
  }
}

/** Anota filas [paquete, acción, detalle, correo] en «Salón · Actualizaciones». */
function registrarActualizaciones_(ss, filas) {
  if (!filas.length) return;
  let hoja = ss.getSheetByName(HOJA_ACTUALIZACIONES);
  if (!hoja) {
    hoja = ss.insertSheet(HOJA_ACTUALIZACIONES, ss.getSheets().length);
    const enc = hoja.getRange(1, 1, 1, 5);
    enc.setValues([['Fecha y hora', 'Paquete', 'Acción', 'Detalle', 'Ingresó']]);
    estiloEncabezado_(enc);
    intentar_('formato del registro', function () {
      hoja.setFrozenRows(1);
      [140, 150, 200, 640, 240].forEach(function (px, j) { hoja.setColumnWidth(j + 1, px); });
      hoja.setTabColor(COLOR_SALON);
    });
    hoja.protect().setDescription('Registro de actualizaciones automáticas').setWarningOnly(true);
  }
  const ahora = new Date();
  const datos = filas.map(function (f) { return [ahora].concat(f.map(function (x) { return celdaTexto_(String(x)); })); });
  const inicio = hoja.getLastRow() + 1;
  const fin = inicio + datos.length - 1;
  if (fin > hoja.getMaxRows()) hoja.insertRowsAfter(hoja.getMaxRows(), fin - hoja.getMaxRows() + 100);

}

// Auxiliares de las actualizaciones

function quitarProtecciones_(hoja) {
  hoja.getProtections(SpreadsheetApp.ProtectionType.SHEET)
    .concat(hoja.getProtections(SpreadsheetApp.ProtectionType.RANGE))
    .forEach(function (p) { p.remove(); });
}

function asegurarTamano_(hoja, filas, columnas) {
  if (hoja.getMaxRows() < filas) hoja.insertRowsAfter(hoja.getMaxRows(), filas - hoja.getMaxRows());
  if (hoja.getMaxColumns() < columnas) hoja.insertColumnsAfter(hoja.getMaxColumns(), columnas - hoja.getMaxColumns());
}

/** Tramos contiguos de una fila con el mismo valor de una propiedad de estilo: [[valor, 'B5:D5'], …]. */
function tramosFila_(ids, estilos, prop, fila, omitir) {
  const salida = [];
  let actual = null, inicio = 0;
  for (let j = 0; j <= ids.length; j++) {
    let v = j < ids.length ? estilos[ids[j]][prop] : null;
    if (v === omitir || v === null || v === undefined || v === false) v = null;
    if (v !== actual) {
      if (actual !== null) salida.push([actual, rangoA1_(fila, inicio + 1, fila, j)]);
      actual = v;
      inicio = j;
    }
  }
  return salida;
}

function agruparTramos_(mapa, tramos) {
  tramos.forEach(function (t) {
    const clave = String(t[0]);
    (mapa[clave] = mapa[clave] || []).push(t[1]);
  });
}

function aplicarGrupos_(hoja, mapa, etiqueta, fn) {
  Object.keys(mapa).forEach(function (clave) {
    const lista = mapa[clave];
    for (let i = 0; i < lista.length; i += 300) {
      const lote = lista.slice(i, i + 300);
      intentar_(etiqueta, function () { fn(hoja.getRangeList(lote), clave === 'true' ? true : clave); });
    }
  });
}

/** Ejecuta un paso cosmético: si falla, lo anota y sigue (los datos nunca dependen de estos pasos). */
function intentar_(etiqueta, fn) {
  try {
    fn();
  } catch (e) {
    console.warn('Formato omitido (' + etiqueta + '): ' + (e && e.message ? e.message : e));
  }
}

function letraColumna_(n) {
  let s = '';
  while (n > 0) {
    const m = (n - 1) % 26;
    s = String.fromCharCode(65 + m) + s;
    n = Math.floor((n - 1) / 26);
  }
  return s;
}

function rangoA1_(f1, c1, f2, c2) {
  const a = letraColumna_(c1) + f1;
  const b = letraColumna_(c2) + f2;
  return a === b ? a : a + ':' + b;
}

function rangoNumerico_(a1) {
  const m = String(a1).match(/^([A-Z]+)(\d+)(?::([A-Z]+)(\d+))?$/);
  const num = function (letras) {
    let n = 0;
    for (let i = 0; i < letras.length; i++) n = n * 26 + (letras.charCodeAt(i) - 64);
    return n;
  };
  const c1 = num(m[1]), r1 = Number(m[2]);
  const c2 = m[3] ? num(m[3]) : c1, r2 = m[4] ? Number(m[4]) : r1;
  return { a1: a1, r1: r1, c1: c1, r2: r2, c2: c2 };
}
